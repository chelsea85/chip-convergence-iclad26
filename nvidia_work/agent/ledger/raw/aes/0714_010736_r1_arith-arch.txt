// FILE: tlul_cmd_intg_chk.v
module tlul_cmd_intg_chk (
	tl_i,
	err_o
);
	localparam signed [31:0] prim_mubi_pkg_MuBi4Width = 4;
	localparam signed [31:0] tlul_pkg_DataIntgWidth = 7;
	localparam signed [31:0] tlul_pkg_H2DCmdIntgWidth = 7;
	localparam signed [31:0] top_pkg_TL_AUW = 23;
	localparam signed [31:0] tlul_pkg_RsvdWidth = ((top_pkg_TL_AUW - prim_mubi_pkg_MuBi4Width) - tlul_pkg_H2DCmdIntgWidth) - tlul_pkg_DataIntgWidth;
	localparam signed [31:0] top_pkg_TL_AIW = 8;
	localparam signed [31:0] top_pkg_TL_AW = 32;
	localparam signed [31:0] top_pkg_TL_DW = 32;
	localparam signed [31:0] top_pkg_TL_DBW = top_pkg_TL_DW >> 3;
	localparam signed [31:0] top_pkg_TL_SZW = $clog2($clog2(top_pkg_TL_DBW) + 1);
	
	// Total width: 7 + 2 + 8 + 32 + 4 + 32 + 5 + 4 + 7 + 7 + 1 = 109 bits [108:0]
	input wire [108:0] tl_i;
	output wire err_o;

	wire [1:0] err;
	wire data_err;
	
	// Flattened extraction of the command payload for SECDED check.
	// Based on the original tlul_pkg_extract_h2d_cmd_intg function logic:
	// payload[42:39] = tl_i[18:15] (instr_type)
	// payload[38:7]  = tl_i[91:60] (address)
	// payload[6:4]   = tl_i[106:104] (opcode/param - original used 4-bit slice tl[107:104] into 3-bit target)
	// payload[3:0]   = tl_i[59:56] (mask)
	wire [42:0] cmd_payload;
	assign cmd_payload = {
		tl_i[18:15], // instr_type
		tl_i[91:60], // address
		tl_i[106:104], // opcode/param
		tl_i[59:56]  // mask
	};

	// SECDED check for command integrity.
	// The input is the 7-bit integrity field followed by the 57-bit padded payload.
	prim_secded_inv_64_57_dec u_chk(
		.data_i({tl_i[14:8], 14'b0, cmd_payload}),
		.data_o(),
		.syndrome_o(),
		.err_o(err)
	);

	// SECDED check for data integrity.
	// The input is the 7-bit integrity field followed by the 32-bit data.
	tlul_data_integ_dec u_tlul_data_integ_dec(
		.data_intg_i({tl_i[7:1], tl_i[55:24]}),
		.data_err_o(data_err)
	);

	// The error output is active only if the command is valid (tl_i[108]).
	// We use a balanced OR tree for the error signals to minimize logic depth.
	wire combined_err;
	assign combined_err = (err[0] | err[1]) | data_err;
	assign err_o = tl_i[108] & combined_err;

	// Logic to avoid lint warnings for unused bits of tl_i.
	wire unused_tl;
	assign unused_tl = ^tl_i;

endmodule
