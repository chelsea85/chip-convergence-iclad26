// FILE: tlul_cmd_intg_chk.v
module tlul_cmd_intg_chk (
	tl_i,
	err_o
);
	localparam signed [31:0] prim_mubi_pkg_MuBi4Width = 4;
	localparam signed [31:0] tlul_pkg_DataIntgWidth = 7;
	localparam signed [31:0] tlul_pkg_H2DCmdIntgWidth = 7;
	localparam signed [31:0] top_pkg_TL_AUW = 23;
	localparam signed [31:0] tlul_pkg_RsvdWidth = ((top_pkg_TL_AUW - prim_mubi_pkg_MuBi4Width) - tlul_pkg_H2DCmdIntgWidth) - tlul_pkg_DataIntgWidth;
	localparam signed [31:0] top_pkg_TL_AIW = 8;
	localparam signed [31:0] top_pkg_TL_AW = 32;
	localparam signed [31:0] top_pkg_TL_DW = 32;
	localparam signed [31:0] top_pkg_TL_DBW = top_pkg_TL_DW >> 3;
	localparam signed [31:0] top_pkg_TL_SZW = $clog2($clog2(top_pkg_TL_DBW) + 1);
	input wire [((((((7 + top_pkg_TL_SZW) + top_pkg_TL_AIW) + top_pkg_TL_AW) + top_pkg_TL_DBW) + top_pkg_TL_DW) + (((tlul_pkg_RsvdWidth + prim_mubi_pkg_MuBi4Width) + tlul_pkg_H2DCmdIntgWidth) + tlul_pkg_DataIntgWidth)) + 0:0] tl_i;
	output wire err_o;
	wire [1:0] err;
	wire data_err;
	wire [(((prim_mubi_pkg_MuBi4Width + top_pkg_TL_AW) + 3) + top_pkg_TL_DBW) - 1:0] cmd;

	// RTLScout-cutpoints: Declare named intermediate wires for extraction
	localparam int Offset = (((tlul_pkg_RsvdWidth + prim_mubi_pkg_MuBi4Width) + tlul_pkg_H2DCmdIntgWidth) + tlul_pkg_DataIntgWidth);
	wire [top_pkg_TL_AW-1:0] a_address_wire = tl_i[top_pkg_TL_AW + (top_pkg_TL_DBW + (top_pkg_TL_DW + Offset)) -: top_pkg_TL_AW];
	wire [2:0] a_op_param_size_wire = tl_i[6 + (top_pkg_TL_SZW + (top_pkg_TL_AIW + (top_pkg_TL_AW + (top_pkg_TL_DBW + (top_pkg_TL_DW + Offset))))) -: 3];
	wire [top_pkg_TL_DBW-1:0] a_mask_wire = tl_i[top_pkg_TL_DBW + (top_pkg_TL_DW + Offset) -: top_pkg_TL_DBW];
	wire [prim_mubi_pkg_MuBi4Width-1:0] instr_type_wire = tl_i[Offset - 5 -: 4];

	assign cmd = {instr_type_wire, a_address_wire, a_op_param_size_wire, a_mask_wire};

	localparam signed [31:0] tlul_pkg_H2DCmdMaxWidth = 57;
	function automatic [56:0] sv2v_cast_57;
		input reg [56:0] inp;
		sv2v_cast_57 = inp;
	endfunction

	wire [6:0] cmd_intg = tl_i[Offset - 9 -: 7];

	prim_secded_inv_64_57_dec u_chk(
		.data_i({cmd_intg, sv2v_cast_57(cmd)}),
		.data_o(),
		.syndrome_o(),
		.err_o(err)
	);

	localparam signed [31:0] tlul_pkg_DataMaxWidth = 32;
	function automatic [31:0] sv2v_cast_32;
		input reg [31:0] inp;
		sv2v_cast_32 = inp;
	endfunction

	wire [6:0] data_intg = tl_i[Offset - 16 -: 7];
	wire [31:0] a_data = tl_i[top_pkg_TL_DW + Offset -: 32];

	tlul_data_integ_dec u_tlul_data_integ_dec(
		.data_intg_i({data_intg, sv2v_cast_32(a_data)}),
		.data_err_o(data_err)
	);

	// P2-pure-sum-noop: ABC handles simple ORs well, but explicit grouping helps clarity
	wire any_err = (|err) | data_err;
	wire a_valid = tl_i[7 + (top_pkg_TL_SZW + (top_pkg_TL_AIW + (top_pkg_TL_AW + (top_pkg_TL_DBW + (top_pkg_TL_DW + Offset)))))];
	assign err_o = a_valid & any_err;

	wire unused_tl;
	assign unused_tl = |tl_i;
endmodule

					add_rk_sel_o = sv2v_cast_7524F(sv2v_cast_19785(5'b01110));
					key_words_sel_o = sv2v_cast_21340(sv2v_cast_26872(5'b10111));
					round_key_sel_o = sv2v_cast_4C47F(sv2v_cast_14B94(3'b011));
				end
				out_valid_o = 1'b1;
				if (out_ready_i) begin
					key_clear_d_o = 1'b0;
					data_out_clear_d_o = 1'b0;
					aes_cipher_ctrl_ns = sv2v_cast_16A1F(6'b001001);
				end
			end
			sv2v_cast_16A1F(6'b010111): alert_o = 1'b1;
			default: begin
				aes_cipher_ctrl_ns = sv2v_cast_16A1F(6'b010111);
				alert_o = 1'b1;
			end
		endcase
		if ((((mux_sel_err_i || sp_enc_err_i) || rnd_ctr_err_i) || op_err_i) || alert_fatal_i)
			aes_cipher_ctrl_ns = sv2v_cast_16A1F(6'b010111);
	end

	prim_sparse_fsm_flop #(
		.Width(aes_pkg_CipherCtrlStateWidth),
		.ResetValue(sv2v_cast_16A1F(6'b001001)),
		.EnableAlertTriggerSVA(1)
	) u_state_regs(
		.clk_i(clk_i),
		.rst_ni(rst_ni),
		.state_i(aes_cipher_ctrl_ns),
		.state_o(aes_cipher_ctrl_cs)
	);

	always @(posedge clk_i or negedge rst_ni) begin : reg_fsm
		if (!rst_ni) begin
			prng_reseed_done_q <= 1'b0;
			rnd_ctr_q <= 4'd0;
			num_rounds_q <= 4'd0;
			num_rounds_regular_q <= 4'd0;
		end
		else begin
			prng_reseed_done_q <= prng_reseed_done_d;
			rnd_ctr_q <= rnd_ctr_d;
			num_rounds_q <= num_rounds_d;
			num_rounds_regular_q <= num_rounds_regular_d;
		end
	end

	assign rnd_ctr_o = rnd_ctr_q;

	generate
		if (SecSBoxImpl == 32'sd4) begin : gen_cyc_ctr
			always @(posedge clk_i or negedge rst_ni) begin : reg_cyc_ctr
				if (!rst_ni)
					cyc_ctr_q <= 3'd0;
				else
					cyc_ctr_q <= cyc_ctr_d;
			end
			// RTLScout-cutpoints: cyc_ctr_q[2] is equivalent to >= 4 for range [0,4]
			assign cyc_ctr_expr = cyc_ctr_q[2];
		end
		else begin : gen_no_cyc_ctr
			always @(*) cyc_ctr_q = 3'd0;
			assign cyc_ctr_expr = 1'b1;
		end
	endgenerate

	initial _sv2v_0 = 0;
endmodule
