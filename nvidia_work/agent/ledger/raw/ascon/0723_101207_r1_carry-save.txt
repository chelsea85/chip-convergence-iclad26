// FILE: tlul_cmd_intg_chk.v
module tlul_cmd_intg_chk (
	tl_i,
	err_o
);
	localparam signed [31:0] prim_mubi_pkg_MuBi4Width = 4;
	localparam signed [31:0] tlul_pkg_DataIntgWidth = 7;
	localparam signed [31:0] tlul_pkg_H2DCmdIntgWidth = 7;
	localparam signed [31:0] top_pkg_TL_AUW = 23;
	localparam signed [31:0] tlul_pkg_RsvdWidth = ((top_pkg_TL_AUW - prim_mubi_pkg_MuBi4Width) - tlul_pkg_H2DCmdIntgWidth) - tlul_pkg_DataIntgWidth;
	localparam signed [31:0] top_pkg_TL_AIW = 8;
	localparam signed [31:0] top_pkg_TL_AW = 32;
	localparam signed [31:0] top_pkg_TL_DW = 32;
	localparam signed [31:0] top_pkg_TL_DBW = top_pkg_TL_DW >> 3;
	localparam signed [31:0] top_pkg_TL_SZW = $clog2($clog2(top_pkg_TL_DBW) + 1);

	// Total width of tl_i is 109 bits [108:0]
	input wire [108:0] tl_i;
	output wire err_o;

	// [rtlscout-cutpoints] Declare named intermediate wires at logical boundaries
	// to help the synthesis tool partition and optimize the combinatorial paths.
	
	// Integrity bits
	wire [6:0] cmd_intg_bits  = tl_i[21:15];
	wire [6:0] data_intg_bits = tl_i[14:8];
	
	// Command payload extraction (equivalent to tlul_pkg_extract_h2d_cmd_intg)
	// payload[42:39] = tl_i[18:15] (Note: overlap with integrity bits is expected in some TL-UL versions)
	// payload[38:7]  = tl_i[91:60] (Address)
	// payload[6:4]   = tl_i[107:105] (Opcode/Size)
	// payload[3:0]   = tl_i[59:56] (Mask/BE)
	wire [42:0] cmd_payload;
	assign cmd_payload[42:39] = tl_i[18:15];
	assign cmd_payload[38:7]  = tl_i[91:60];
	assign cmd_payload[6:4]   = tl_i[107:105];
	assign cmd_payload[3:0]   = tl_i[59:56];

	// Data payload extraction
	wire [31:0] data_payload = tl_i[55:24];
	
	// Valid bit
	wire tl_valid = tl_i[108];

	// SECDED Decoding for Command
	// The decoder expects 64 bits: {7 bits integrity, 57 bits data}
	// We pad the 43-bit cmd_payload to 57 bits.
	wire [1:0] cmd_err;
	prim_secded_inv_64_57_dec u_chk(
		.data_i({cmd_intg_bits, 14'b0, cmd_payload}),
		.data_o(),
		.syndrome_o(),
		.err_o(cmd_err)
	);

	// SECDED Decoding for Data
	wire data_err;
	tlul_data_integ_dec u_tlul_data_integ_dec(
		.data_intg_i({data_intg_bits, data_payload}),
		.data_err_o(data_err)
	);

	// Final error output
	// [rtlscout-cutpoints] Use intermediate wires for the reduction ORs
	wire combined_err = (|cmd_err) | data_err;
	assign err_o = tl_valid ? combined_err : 1'b0;

	// Dummy to avoid unused wire warnings if any bits were skipped in manual extraction
	wire _unused = &{tl_i};

endmodule
