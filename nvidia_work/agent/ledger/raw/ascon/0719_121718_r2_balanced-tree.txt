// FILE: prim_subreg_shadow.v
module prim_subreg_shadow (
	clk_i,
	rst_ni,
	rst_shadowed_ni,
	re,
	we,
	wd,
	de,
	d,
	qe,
	q,
	ds,
	qs,
	phase,
	err_update,
	err_storage
);
	reg _sv2v_0;
	parameter signed [31:0] DW = 32;
	parameter [2:0] SwAccess = 3'd0;
	parameter [DW - 1:0] RESVAL = 1'sb0;
	parameter [0:0] Mubi = 1'b0;
	input clk_i;
	input rst_ni;
	input rst_shadowed_ni;
	input re;
	input we;
	input [DW - 1:0] wd;
	input de;
	input [DW - 1:0] d;
	output wire qe;
	output wire [DW - 1:0] q;
	output wire [DW - 1:0] ds;
	output wire [DW - 1:0] qs;
	output wire phase;
	output wire err_update;
	output wire err_storage;

	localparam [2:0] InvertedSwAccess = (SwAccess == 3'd4 ? 3'd5 : (SwAccess == 3'd5 ? 3'd4 : SwAccess));

	wire phase_clear;
	reg phase_q;
	reg shadow_we;
	reg [DW - 1:0] shadow_wd;
	reg [DW - 1:0] shadow_q;
	wire [DW - 1:0] committed_q;
	wire [DW - 1:0] committed_qs;
	wire wr_en;
	wire [DW - 1:0] wr_data;

	// [rtlscout-cutpoints] Use named wires for wide comparison results to help synthesis partitioning
	wire shadow_match_committed;
	wire shadow_match_wr_data;
	wire shadow_match_wd;

	prim_subreg_arb #(
		.DW(DW),
		.SwAccess(SwAccess),
		.Mubi(Mubi)
	) wr_en_data_arb(
		.we(we),
		.wd(wd),
		.de(de),
		.d(d),
		.q(q),
		.wr_en(wr_en),
		.wr_data(wr_data)
	);

	assign phase_clear = (SwAccess == 3'd1 ? 1'b0 : re);

	// [balanced-tree] Re-organize error and phase logic into a more parallel structure
	// shadow_match_committed is true if shadow_q is the bitwise inverse of committed_q
	assign shadow_match_committed = &(shadow_q ^ committed_q);
	// shadow_match_wr_data is true if shadow_q is the bitwise inverse of wr_data
	assign shadow_match_wr_data = &(shadow_q ^ wr_data);
	// shadow_match_wd is true if shadow_q is the bitwise inverse of wd
	assign shadow_match_wd = &(shadow_q ^ wd);

	assign err_storage = !phase_q && !shadow_match_committed;

	wire any_err;
	wire phase_d;

	always @(posedge clk_i or negedge rst_ni) begin : phase_reg
		if (!rst_ni)
			phase_q <= 1'b0;
		else
			phase_q <= phase_d;
	end

	always @(posedge clk_i or negedge rst_shadowed_ni)
		if (!rst_shadowed_ni)
			shadow_q <= ~RESVAL;
		else if (shadow_we)
			shadow_q <= shadow_wd;

	generate
		if (InvertedSwAccess == SwAccess) begin : gen_shadow_reg_std
			wire shadow_wr_en;
			wire [DW - 1:0] shadow_wr_data;
			prim_subreg_arb #(
				.DW(DW),
				.SwAccess(InvertedSwAccess),
				.Mubi(Mubi)
			) wr_en_data_arb_shadow(
				.we(we),
				.wd(~wr_data),
				.de(de),
				.d(~d),
				.q(shadow_q),
				.wr_en(shadow_wr_en),
				.wr_data(shadow_wr_data)
			);

			assign err_update = phase_q && wr_en && !shadow_match_wr_data;
			assign any_err = err_storage | err_update;
			assign phase_d = (phase_clear || any_err) ? 1'b0 : (wr_en ? ~phase_q : phase_q);

			always @(*) begin
				shadow_we = 1'b0;
				shadow_wd = shadow_wr_data;
				if (!err_storage) begin
					if (err_update || phase_clear) begin
						shadow_we = 1'b1;
						shadow_wd = ~committed_q;
					end
					else if (!phase_q && shadow_wr_en) begin
						shadow_we = 1'b1;
						shadow_wd = shadow_wr_data;
					end
				end
			end
		end
		else begin : gen_shadow_reg_wxx
			wire shadow_wr_en_phase0;
			wire shadow_wr_en_phase1;
			wire [DW - 1:0] shadow_wr_data_phase0;
			wire [DW - 1:0] shadow_wr_data_phase1;
			prim_subreg_arb #(
				.DW(DW),
				.SwAccess(3'd0),
				.Mubi(Mubi)
			) wr_en_data_arb_phase0(
				.we(we),
				.wd(~wd),
				.de(de),
				.d(~d),
				.q(shadow_q),
				.wr_en(shadow_wr_en_phase0),
				.wr_data(shadow_wr_data_phase0)
			);
			prim_subreg_arb #(
				.DW(DW),
				.SwAccess(InvertedSwAccess),
				.Mubi(Mubi)
			) wr_en_data_arb_phase1(
				.we(we),
				.wd(~wr_data),
				.de(de),
				.d(~d),
				.q(~committed_q),
				.wr_en(shadow_wr_en_phase1),
				.wr_data(shadow_wr_data_phase1)
			);

			assign err_update = phase_q && wr_en && !shadow_match_wd;
			assign any_err = err_storage | err_update;
			assign phase_d = (phase_clear || any_err) ? 1'b0 : (wr_en ? ~phase_q : phase_q);

			always @(*) begin
				shadow_we = 1'b0;
				shadow_wd = shadow_wr_data_phase0;
				if (!err_storage) begin
					if (err_update || phase_clear) begin
						shadow_we = 1'b1;
						shadow_wd = ~committed_q;
					end
					else if (!phase_q) begin
						shadow_we = shadow_wr_en_phase0;
						shadow_wd = shadow_wr_data_phase0;
					end
					else begin
						shadow_we = shadow_wr_en_phase1;
						shadow_wd = shadow_wr_data_phase1;
					end
				end
			end
		end
	endgenerate

	wire committed_we = phase_q && wr_en && !any_err && !err_storage;
	wire committed_de = phase_q && de && !any_err && !err_storage;

	prim_subreg #(
		.DW(DW),
		.SwAccess(SwAccess),
		.RESVAL(RESVAL),
		.Mubi(Mubi)
	) committed_reg(
		.clk_i(clk_i),
		.rst_ni(rst_ni),
		.we(committed_we),
		.wd(wr_data),
		.de(committed_de),
		.d(d),
		.qe(qe),
		.q(committed_q),
		.ds(ds),
		.qs(committed_qs)
	);

	assign phase = phase_q;
	assign q = committed_q;
	assign qs = committed_qs;
	initial _sv2v_0 = 0;
endmodule
