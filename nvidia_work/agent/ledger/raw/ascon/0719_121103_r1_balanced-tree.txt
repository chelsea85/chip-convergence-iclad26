// FILE: tlul_cmd_intg_chk.v
module tlul_cmd_intg_chk (
	tl_i,
	err_o
);
	localparam signed [31:0] prim_mubi_pkg_MuBi4Width = 4;
	localparam signed [31:0] tlul_pkg_DataIntgWidth = 7;
	localparam signed [31:0] tlul_pkg_H2DCmdIntgWidth = 7;
	localparam signed [31:0] top_pkg_TL_AUW = 23;
	localparam signed [31:0] tlul_pkg_RsvdWidth = ((top_pkg_TL_AUW - prim_mubi_pkg_MuBi4Width) - tlul_pkg_H2DCmdIntgWidth) - tlul_pkg_DataIntgWidth;
	localparam signed [31:0] top_pkg_TL_AIW = 8;
	localparam signed [31:0] top_pkg_TL_AW = 32;
	localparam signed [31:0] top_pkg_TL_DW = 32;
	localparam signed [31:0] top_pkg_TL_DBW = top_pkg_TL_DW >> 3;
	localparam signed [31:0] top_pkg_TL_SZW = $clog2($clog2(top_pkg_TL_DBW) + 1);
	
	localparam signed [31:0] TotalWidth = ((((((7 + top_pkg_TL_SZW) + top_pkg_TL_AIW) + top_pkg_TL_AW) + top_pkg_TL_DBW) + top_pkg_TL_DW) + (((tlul_pkg_RsvdWidth + prim_mubi_pkg_MuBi4Width) + tlul_pkg_H2DCmdIntgWidth) + tlul_pkg_DataIntgWidth)) + 0;
	
	input wire [TotalWidth:0] tl_i;
	output wire err_o;

	wire [1:0] err;
	wire data_err;
	
	// [rtlscout-cutpoints] Use named intermediate wires to help Yosys/ABC partition logic.
	localparam signed [31:0] CmdWidth = (((prim_mubi_pkg_MuBi4Width + top_pkg_TL_AW) + 3) + top_pkg_TL_DBW);
	wire [CmdWidth - 1:0] cmd;

	// Manual extraction of fields to avoid complex function overhead and deep logic chains.
	// The indices are derived from the tlul_pkg structure.
	// Base offset for integrity fields
	localparam int IntgBase = 0;
	localparam int DataIntgBase = IntgBase + tlul_pkg_DataIntgWidth;
	localparam int CmdIntgBase = DataIntgBase + tlul_pkg_H2DCmdIntgWidth;
	localparam int MuBiBase = CmdIntgBase + prim_mubi_pkg_MuBi4Width;
	localparam int RsvdBase = MuBiBase + tlul_pkg_RsvdWidth;
	
	// Base offset for payload fields
	localparam int DataBase = RsvdBase;
	localparam int MaskBase = DataBase + top_pkg_TL_DW;
	localparam int AddrBase = MaskBase + top_pkg_TL_DBW;
	localparam int IdBase   = AddrBase + top_pkg_TL_AW;
	localparam int SizeBase = IdBase + top_pkg_TL_AIW;
	localparam int OpBase   = SizeBase + top_pkg_TL_SZW;
	localparam int ValidBase = OpBase + 3;

	// cmd = {instr_type, a_address, a_size, a_param, a_opcode, a_mask}
	// Note: prim_mubi_pkg_MuBi4Width is used for instr_type (4 bits)
	assign cmd[CmdWidth-1 : CmdWidth-4] = tl_i[CmdIntgBase + 3 : CmdIntgBase];
	assign cmd[CmdWidth-5 : top_pkg_TL_DBW+3] = tl_i[AddrBase + top_pkg_TL_AW - 1 : AddrBase];
	assign cmd[top_pkg_TL_DBW+2 : top_pkg_TL_DBW] = tl_i[OpBase + 2 : OpBase];
	assign cmd[top_pkg_TL_DBW-1 : 0] = tl_i[MaskBase + top_pkg_TL_DBW - 1 : MaskBase];

	prim_secded_inv_64_57_dec u_chk(
		.data_i({tl_i[DataIntgBase + tlul_pkg_H2DCmdIntgWidth - 1 : DataIntgBase], 7'b0, cmd}),
		.data_o(),
		.syndrome_o(),
		.err_o(err)
	);

	tlul_data_integ_dec u_tlul_data_integ_dec(
		.data_intg_i({tl_i[IntgBase + tlul_pkg_DataIntgWidth - 1 : IntgBase], tl_i[DataBase + top_pkg_TL_DW - 1 : DataBase]}),
		.data_err_o(data_err)
	);

	// [p2-pure-sum-noop] Re-associating the reduction ORs into a balanced tree.
	wire combined_err;
	assign combined_err = (err[0] | err[1]) | data_err;
	assign err_o = tl_i[ValidBase] & combined_err;

endmodule

// FILE: prim_subreg_shadow.v
module prim_subreg_shadow (
	clk_i,
	rst_ni,
	rst_shadowed_ni,
	re,
	we,
	wd,
	de,
	d,
	qe,
	q,
	ds,
	qs,
	phase,
	err_update,
	err_storage
);
	reg _sv2v_0;
	parameter signed [31:0] DW = 32;
	parameter [2:0] SwAccess = 3'd0;
	parameter [DW - 1:0] RESVAL = 1'sb0;
	parameter [0:0] Mubi = 1'b0;
	input clk_i;
	input rst_ni;
	input rst_shadowed_ni;
	input re;
	input we;
	input [DW - 1:0] wd;
	input de;
	input [DW - 1:0] d;
	output wire qe;
	output wire [DW - 1:0] q;
	output wire [DW - 1:0] ds;
	output wire [DW - 1:0] qs;
	output wire phase;
	output wire err_update;
	output wire err_storage;

	localparam [2:0] InvertedSwAccess = (SwAccess == 3'd4 ? 3'd5 : (SwAccess == 3'd5 ? 3'd4 : SwAccess));
	wire phase_clear;
	reg phase_q;
	reg shadow_we;
	wire committed_we;
	wire committed_de;
	wire committed_qe;
	reg [DW - 1:0] shadow_wd;
	reg [DW - 1:0] shadow_q;
	wire [DW - 1:0] committed_q;
	wire [DW - 1:0] committed_qs;
	wire wr_en;
	wire [DW - 1:0] wr_data;

	prim_subreg_arb #(
		.DW(DW),
		.SwAccess(SwAccess),
		.Mubi(Mubi)
	) wr_en_data_arb(
		.we(we),
		.wd(wd),
		.de(de),
		.d(d),
		.q(q),
		.wr_en(wr_en),
		.wr_data(wr_data)
	);

	assign phase_clear = (SwAccess == 3'd1 ? 1'b0 : re);

	// [rtlscout-cutpoints] Use named wires for error conditions to help timing.
	wire update_fail;
	wire storage_fail;

	always @(posedge clk_i or negedge rst_ni) begin : phase_reg
		if (!rst_ni)
			phase_q <= 1'b0;
		else if ((wr_en && !err_storage) && !err_update)
			phase_q <= ~phase_q;
		else if ((phase_clear || err_storage) || err_update)
			phase_q <= 1'b0;
	end

	always @(posedge clk_i or negedge rst_shadowed_ni)
		if (!rst_shadowed_ni)
			shadow_q <= ~RESVAL;
		else if (shadow_we)
			shadow_q <= shadow_wd;

	generate
		if (InvertedSwAccess == SwAccess) begin : gen_shadow_reg_std
			wire shadow_wr_en;
			wire [DW - 1:0] shadow_wr_data;
			prim_subreg_arb #(
				.DW(DW),
				.SwAccess(InvertedSwAccess),
				.Mubi(Mubi)
			) wr_en_data_arb_shadow(
				.we(we),
				.wd(~wr_data),
				.de(de),
				.d(~d),
				.q(shadow_q),
				.wr_en(shadow_wr_en),
				.wr_data(shadow_wr_data)
			);
			always @(*) begin
				shadow_we = 1'b0;
				shadow_wd = shadow_wr_data;
				if (!err_storage) begin
					if (err_update || phase_clear) begin
						shadow_we = 1'b1;
						shadow_wd = ~committed_q;
					end
					else if (!phase_q && shadow_wr_en) begin
						shadow_we = 1'b1;
						shadow_wd = shadow_wr_data;
					end
				end
			end
			assign update_fail = |(shadow_q ^ ~wr_data);
			assign err_update = update_fail & phase_q & wr_en;
		end
		else begin : gen_shadow_reg_wxx
			wire shadow_wr_en_phase0;
			wire shadow_wr_en_phase1;
			wire [DW - 1:0] shadow_wr_data_phase0;
			wire [DW - 1:0] shadow_wr_data_phase1;
			prim_subreg_arb #(
				.DW(DW),
				.SwAccess(3'd0),
				.Mubi(Mubi)
			) wr_en_data_arb_phase0(
				.we(we),
				.wd(~wd),
				.de(de),
				.d(~d),
				.q(shadow_q),
				.wr_en(shadow_wr_en_phase0),
				.wr_data(shadow_wr_data_phase0)
			);
			prim_subreg_arb #(
				.DW(DW),
				.SwAccess(InvertedSwAccess),
				.Mubi(Mubi)
			) wr_en_data_arb_phase1(
				.we(we),
				.wd(~wr_data),
				.de(de),
				.d(~d),
				.q(~committed_q),
				.wr_en(shadow_wr_en_phase1),
				.wr_data(shadow_wr_data_phase1)
			);
			always @(*) begin
				shadow_we = 1'b0;
				shadow_wd = shadow_wr_data_phase0;
				if (!err_storage) begin
					if (err_update || phase_clear) begin
						shadow_we = 1'b1;
						shadow_wd = ~committed_q;
					end
					else if (!phase_q) begin
						shadow_we = shadow_wr_en_phase0;
						shadow_wd = shadow_wr_data_phase0;
					end
					else begin
						shadow_we = shadow_wr_en_phase1;
						shadow_wd = shadow_wr_data_phase1;
					end
				end
			end
			assign update_fail = |(shadow_q ^ ~wd);
			assign err_update = phase_q & wr_en & update_fail;
		end
	endgenerate

	assign committed_we = ((we & phase_q) & ~err_update) & ~err_storage;
	assign committed_de = ((de & phase_q) & ~err_update) & ~err_storage;

	prim_subreg #(
		.DW(DW),
		.SwAccess(SwAccess),
		.RESVAL(RESVAL),
		.Mubi(Mubi)
	) committed_reg(
		.clk_i(clk_i),
		.rst_ni(rst_ni),
		.we(committed_we),
		.wd(wr_data),
		.de(committed_de),
		.d(d),
		.qe(committed_qe),
		.q(committed_q),
		.ds(ds),
		.qs(committed_qs)
	);

	assign phase = phase_q;
	assign storage_fail = |(shadow_q ^ ~committed_q);
	assign err_storage = storage_fail & ~phase_q;
	assign qe = committed_qe;
	assign q = committed_q;
	assign qs = committed_qs;
	initial _sv2v_0 = 0;
endmodule
