To optimize the PPA of the SHA-512 design, we can clean up the dead code left over from previous optimizations (specifically the `t1` and `t2` signals and their associated logic, which were replaced by the carry-save adder trees `a_new_csa` and `e_new_csa`). The carry-save adder (CSA) trees are already correctly implemented and effectively compress the 3+ operand additions on the critical path into a single carry-propagate addition, which is handled by the Kogge-Stone adder for `a_new` and a standard `+` for `e_new`.

Here is the cleaned-up `sha512_core.v` file:

`default_nettype none

module sha512_core(
                   input wire            clk,
                   input wire            reset_n,

                   input wire            init,
                   input wire            next,
                   input wire [1 : 0]    mode,

                   input wire            work_factor,
                   input wire [31 : 0]   work_factor_num,

                   input wire [1023 : 0] block,

                   output wire           ready,
                   output wire [511 : 0] digest,
                   output wire           digest_valid
                  );


  //----------------------------------------------------------------
  // Internal constant and parameter definitions.
  //----------------------------------------------------------------
  localparam SHA512_ROUNDS = 79;

  localparam CTRL_IDLE   = 2'h0;
  localparam CTRL_ROUNDS = 2'h1;
  localparam CTRL_DONE   = 2'h2;


  //----------------------------------------------------------------
  // Registers including update variables and write enable.
  //----------------------------------------------------------------
  reg [63 : 0] a_reg;
  reg [63 : 0] a_new;
  reg [63 : 0] b_reg;
  reg [63 : 0] b_new;
  reg [63 : 0] c_reg;
  reg [63 : 0] c_new;
  reg [63 : 0] d_reg;
  reg [63 : 0] d_new;
  reg [63 : 0] e_reg;
  reg [63 : 0] e_new;
  reg [63 : 0] f_reg;
  reg [63 : 0] f_new;
  reg [63 : 0] g_reg;
  reg [63 : 0] g_new;
  reg [63 : 0] h_reg;
  reg [63 : 0] h_new;
  reg          a_h_we;

  reg [63 : 0] H0_reg;
  reg [63 : 0] H0_new;
  reg [63 : 0] H1_reg;
  reg [63 : 0] H1_new;
  reg [63 : 0] H2_reg;
  reg [63 : 0] H2_new;
  reg [63 : 0] H3_reg;
  reg [63 : 0] H3_new;
  reg [63 : 0] H4_reg;
  reg [63 : 0] H4_new;
  reg [63 : 0] H5_reg;
  reg [63 : 0] H5_new;
  reg [63 : 0] H6_reg;
  reg [63 : 0] H6_new;
  reg [63 : 0] H7_reg;
  reg [63 : 0] H7_new;
  reg          H_we;

  reg [6 : 0]  round_ctr_reg;
  reg [6 : 0]  round_ctr_new;
  reg          round_ctr_we;
  reg          round_ctr_inc;
  reg          round_ctr_rst;

  reg [31 : 0] work_factor_ctr_reg;
  reg [31 : 0] work_factor_ctr_new;
  reg          work_factor_ctr_rst;
  reg          work_factor_ctr_inc;
  reg          work_factor_ctr_we;

  reg          ready_reg;
  reg          ready_new;
  reg          ready_we;

  reg          digest_valid_reg;
  reg          digest_valid_new;
  reg          digest_valid_we;

  reg [1 : 0]  sha512_ctrl_reg;
  reg [1 : 0]  sha512_ctrl_new;
  reg          sha512_ctrl_we;


  //----------------------------------------------------------------
  // Wires.
  //----------------------------------------------------------------
  reg digest_init;
  reg digest_update;

  reg state_init;
  reg state_update;

  reg first_block;

  // Round helper terms (combinational), exposed at module scope so the state
  // additions below can be written as BALANCED adder trees. Modular (mod 2^64)
  // addition is associative, so regrouping is value-identical but shortens the
  // critical path (depth ~3 vs the original left-associative ~6 chain).
  wire [63 : 0] sum0_w = {a_reg[27:0], a_reg[63:28]} ^ {a_reg[33:0], a_reg[63:34]} ^ {a_reg[38:0], a_reg[63:39]};
  wire [63 : 0] sum1_w = {e_reg[13:0], e_reg[63:14]} ^ {e_reg[17:0], e_reg[63:18]} ^ {e_reg[40:0], e_reg[63:41]};
  wire [63 : 0] ch_w   = (e_reg & f_reg) ^ ((~e_reg) & g_reg);
  wire [63 : 0] maj_w  = (a_reg & b_reg) ^ (a_reg & c_reg) ^ (b_reg & c_reg);

  wire [63 : 0] k_data;

  reg           w_init;
  reg           w_next;
  wire [63 : 0] w_data;

  wire [63 : 0] H0_0;
  wire [63 : 0] H0_1;
  wire [63 : 0] H0_2;
  wire [63 : 0] H0_3;
  wire [63 : 0] H0_4;
  wire [63 : 0] H0_5;
  wire [63 : 0] H0_6;
  wire [63 : 0] H0_7;


  //----------------------------------------------------------------
  // Carry-Save Adder (CSA) Trees for Round Logic
  //----------------------------------------------------------------
  function [127:0] csa32;
    input [63:0] x;
    input [63:0] y;
    input [63:0] z;
    begin
      csa32[63:0]   = x ^ y ^ z;
      csa32[127:64] = ((x & y) | (y & z) | (z & x)) << 1;
    end
  endfunction

  // a_new CSA tree (7 terms: h_reg, sum1_w, ch_w, sum0_w, maj_w, k_data, w_data)
  // Structured to compress early-arriving signals first, injecting late-arriving
  // signals (k_data, w_data) closer to the final carry-propagate adder.
  wire [127:0] sa0_0_csa = csa32(h_reg, sum1_w, ch_w);
  wire [63:0]  sa0_0     = sa0_0_csa[63:0];
  wire [63:0]  ca0_0     = sa0_0_csa[127:64];

  wire [127:0] sa1_0_csa = csa32(sa0_0, ca0_0, sum0_w);
  wire [63:0]  sa1_0     = sa1_0_csa[63:0];
  wire [63:0]  ca1_0     = sa1_0_csa[127:64];

  wire [127:0] sa2_0_csa = csa32(sa1_0, ca1_0, maj_w);
  wire [63:0]  sa2_0     = sa2_0_csa[63:0];
  wire [63:0]  ca2_0     = sa2_0_csa[127:64];

  wire [127:0] sa3_0_csa = csa32(sa2_0, ca2_0, k_data);
  wire [63:0]  sa3_0     = sa3_0_csa[63:0];
  wire [63:0]  ca3_0     = sa3_0_csa[127:64];

  wire [127:0] sa4_0_csa = csa32(sa3_0, ca3_0, w_data);
  wire [63:0]  sa4_0     = sa4_0_csa[63:0];
  wire [63:0]  ca4_0     = sa4_0_csa[127:64];

  // Swapped default '+' operator with a high-performance Kogge-Stone parallel-prefix adder
  wire [63:0]  a_new_csa;
  kogge_stone_64 adder_a (
    .a(sa4_0),
    .b(ca4_0),
    .sum(a_new_csa)
  );

  // e_new CSA tree (6 terms: d_reg, h_reg, sum1_w, ch_w, k_data, w_data)
  // Structured similarly to optimize for late-arriving k_data and w_data.
  wire [127:0] se0_0_csa = csa32(d_reg, h_reg, sum1_w);
  wire [63:0]  se0_0     = se0_0_csa[63:0];
  wire [63:0]  ce0_0     = se0_0_csa[127:64];

  wire [127:0] se1_0_csa = csa32(se0_0, ce0_0, ch_w);
  wire [63:0]  se1_0     = se1_0_csa[63:0];
  wire [63:0]  ce1_0     = se1_0_csa[127:64];

  wire [127:0] se2_0_csa = csa32(se1_0, ce1_0, k_data);
  wire [63:0]  se2_0     = se2_0_csa[63:0];
  wire [63:0]  ce2_0     = se2_0_csa[127:64];

  wire [127:0] se3_0_csa = csa32(se2_0, ce2_0, w_data);
  wire [63:0]  se3_0     = se3_0_csa[63:0];
  wire [63:0]  ce3_0     = se3_0_csa[127:64];

  wire [63:0]  e_new_csa = se3_0 + ce3_0;


  //----------------------------------------------------------------
  // Module instantiantions.
  //----------------------------------------------------------------
  sha512_k_constants k_constants_inst(
                                      .addr(round_ctr_reg),
                                      .K(k_data)
                                     );


  sha512_h_constants h_constants_inst(
                                      .mode(mode),

                                      .H0(H0_0),
                                      .H1(H0_1),
                                      .H2(H0_2),
                                      .H3(H0_3),
                                      .H4(H0_4),
                                      .H5(H0_5),
                                      .H6(H0_6),
                                      .H7(H0_7)
                                     );


  sha512_w_mem w_mem_inst(
                          .clk(clk),
                          .reset_n(reset_n),

                          .block(block),

                          .init(w_init),
                          .next(w_next),
                          .w(w_data)
                         );


  //----------------------------------------------------------------
  // Concurrent connectivity for ports etc.
  //----------------------------------------------------------------
  assign ready = ready_reg;

  assign digest = {H0_reg, H1_reg, H2_reg, H3_reg,
                   H4_reg, H5_reg, H6_reg, H7_reg};

  assign digest_valid = digest_valid_reg;


  //----------------------------------------------------------------
  // reg_update
  // Update functionality for all registers in the core.
  // All registers are positive edge triggered with asynchronous
  // active low reset. All registers have write enable.
  //----------------------------------------------------------------
  always @ (posedge clk or negedge reset_n)
    begin : reg_update
      if (!reset_n)
        begin
          a_reg               <= 64'h0;
          b_reg               <= 64'h0;
          c_reg               <= 64'h0;
          d_reg               <= 64'h0;
          e_reg               <= 64'h0;
          f_reg               <= 64'h0;
          g_reg               <= 64'h0;
          h_reg               <= 64'h0;
          H0_reg              <= 64'h0;
          H1_reg              <= 64'h0;
          H2_reg              <= 64'h0;
          H3_reg              <= 64'h0;
          H4_reg              <= 64'h0;
          H5_reg              <= 64'h0;
          H6_reg              <= 64'h0;
          H7_reg              <= 64'h0;
          work_factor_ctr_reg <= 32'h0;
          ready_reg           <= 1'b1;
          digest_valid_reg    <= 1'b0;
          round_ctr_reg       <= 7'h0;
          sha512_ctrl_reg     <= CTRL_IDLE;
        end

      else
        begin
          if (a_h_we)
            begin
              a_reg <= a_new;
              b_reg <= b_new;
              c_reg <= c_new;
              d_reg <= d_new;
              e_reg <= e_new;
              f_reg <= f_new;
              g_reg <= g_new;
              h_reg <= h_new;
            end

          if (H_we)
            begin
              H0_reg <= H0_new;
              H1_reg <= H1_new;
              H2_reg <= H2_new;
              H3_reg <= H3_new;
              H4_reg <= H4_new;
              H5_reg <= H5_new;
              H6_reg <= H6_new;
              H7_reg <= H7_new;
            end

          if (round_ctr_we)
            round_ctr_reg <= round_ctr_new;

          if (work_factor_ctr_we)
            work_factor_ctr_reg <= work_factor_ctr_new;

          if (ready_we)
            ready_reg <= ready_new;

          if (digest_valid_we)
            digest_valid_reg <= digest_valid_new;

          if (sha512_ctrl_we)
            sha512_ctrl_reg <= sha512_ctrl_new;
        end
    end // reg_update


  //----------------------------------------------------------------
  // digest_logic
  //
  // The logic needed to init as well as update the digest.
  //----------------------------------------------------------------
  always @*
    begin : digest_logic
      H0_new = 64'h0;
      H1_new = 64'h0;
      H2_new = 64'h0;
      H3_new = 64'h0;
      H4_new = 64'h0;
      H5_new = 64'h0;
      H6_new = 64'h0;
      H7_new = 64'h0;
      H_we = 0;

      if (digest_init)
        begin
          H0_new = H0_0;
          H1_new = H0_1;
          H2_new = H0_2;
          H3_new = H0_3;
          H4_new = H0_4;
          H5_new = H0_5;
          H6_new = H0_6;
          H7_new = H0_7;
          H_we = 1;
        end

      if (digest_update)
        begin
          H0_new = H0_reg + a_reg;
          H1_new = H1_reg + b_reg;
          H2_new = H2_reg + c_reg;
          H3_new = H3_reg + d_reg;
          H4_new = H4_reg + e_reg;
          H5_new = H5_reg + f_reg;
          H6_new = H6_reg + g_reg;
          H7_new = H7_reg + h_reg;
          H_we = 1;
        end
    end // digest_logic


  //----------------------------------------------------------------
  // state_logic
  //
  // The logic needed to init as well as update the state during
  // round processing.
  //----------------------------------------------------------------
  always @*
    begin : state_logic
      a_new  = 64'h0;
      b_new  = 64'h0;
      c_new  = 64'h0;
      d_new  = 64'h0;
      e_new  = 64'h0;
      f_new  = 64'h0;
      g_new  = 64'h0;
      h_new  = 64'h0;
      a_h_we = 0;

      if (state_init)
        begin
          if (first_block)
            begin
              a_new  = H0_0;
              b_new  = H0_1;
              c_new  = H0_2;
              d_new  = H0_3;
              e_new  = H0_4;
              f_new  = H0_5;
              g_new  = H0_6;
              h_new  = H0_7;
              a_h_we = 1;
            end
          else
            begin
              a_new  = H0_reg;
              b_new  = H1_reg;
              c_new  = H2_reg;
              d_new  = H3_reg;
              e_new  = H4_reg;
              f_new  = H5_reg;
              g_new  = H6_reg;
              h_new  = H7_reg;
              a_h_we = 1;
            end
        end

      if (state_update)
        begin
          // Optimized using Carry-Save Adder (CSA) trees to minimize critical path.
          a_new  = a_new_csa;
          b_new  = a_reg;
          c_new  = b_reg;
          d_new  = c_reg;
          e_new  = e_new_csa;
          f_new  = e_reg;
          g_new  = f_reg;
          h_new  = g_reg;
          a_h_we = 1;
        end
    end // state_logic


  //----------------------------------------------------------------
  // round_ctr
  //
  // Update logic for the round counter, a monotonically
  // increasing counter with reset.
  //----------------------------------------------------------------
  always @*
    begin : round_ctr
      round_ctr_new = 7'h00;
      round_ctr_we  = 0;

      if (round_ctr_rst)
        begin
          round_ctr_new = 7'h00;
          round_ctr_we  = 1;
        end

      if (round_ctr_inc)
        begin
          round_ctr_new = round_ctr_reg + 1'b1;
          round_ctr_we  = 1;
        end
    end // round_ctr


  //----------------------------------------------------------------
  // work_factor_ctr
  //
  // Work factor counter logic.
  //----------------------------------------------------------------
  always @*
    begin : work_factor_ctr
      work_factor_ctr_new  = 32'h0;
      work_factor_ctr_we   = 0;

      if (work_factor_ctr_rst)
        begin
          work_factor_ctr_new = 32'h0;
          work_factor_ctr_we  = 1;
        end

      if (work_factor_ctr_inc)
        begin
          work_factor_ctr_new = work_factor_ctr_reg + 1'b1;
          work_factor_ctr_we  = 1;
        end
    end // work_factor_ctr


  //----------------------------------------------------------------
  // sha512_ctrl_fsm
  //
  // Logic for the state machine controlling the core behaviour.
  //----------------------------------------------------------------
  always @*
    begin : sha512_ctrl_fsm
      digest_init         = 1'b0;
      digest_update       = 1'b0;
      state_init          = 1'b0;
      state_update        = 1'b0;
      first_block         = 1'b0;
      w_init              = 1'b0;
      w_next              = 1'b0;
      round_ctr_inc       = 1'b0;
      round_ctr_rst       = 1'b0;
      digest_valid_new    = 1'b0;
      digest_valid_we     = 1'b0;
      work_factor_ctr_rst = 1'b0;
      work_factor_ctr_inc = 1'b0;
      ready_new           = 1'b0;
      ready_we            = 1'b0;
      sha512_ctrl_new     = CTRL_IDLE;
      sha512_ctrl_we      = 1'b0;

      case (sha512_ctrl_reg)
        CTRL_IDLE:
          begin
            if (init)
              begin
                ready_new           = 1'b0;
                ready_we            = 1'b1;
                work_factor_ctr_rst = 1;
                digest_init         = 1;
                w_init              = 1;
                state_init          = 1;
                first_block         = 1;
                round_ctr_rst       = 1;
                digest_valid_new    = 0;
                digest_valid_we     = 1;
                sha512_ctrl_new     = CTRL_ROUNDS;
                sha512_ctrl_we      = 1;
              end

            if (next)
              begin
                ready_new           = 1'b0;
                ready_we            = 1'b1;
                work_factor_ctr_rst = 1;
                w_init              = 1;
                state_init          = 1;
                round_ctr_rst       = 1;
                digest_valid_new    = 0;
                digest_valid_we     = 1;
                sha512_ctrl_new     = CTRL_ROUNDS;
                sha512_ctrl_we      = 1;
              end
          end


        CTRL_ROUNDS:
          begin
            w_next        = 1;
            state_update  = 1;
            round_ctr_inc = 1;

            if (round_ctr_reg == SHA512_ROUNDS)
              begin
                work_factor_ctr_inc = 1;
                sha512_ctrl_new     = CTRL_DONE;
                sha512_ctrl_we      = 1;
              end
          end


        CTRL_DONE:
          begin
            if (work_factor)
              begin
                if (work_factor_ctr_reg < work_factor_num)
                  begin
                    w_init              = 1'b1;
                    state_init          = 1'b1;
                    round_ctr_rst       = 1'b1;
                    sha512_ctrl_new     = CTRL_ROUNDS;
                    sha512_ctrl_we      = 1'b1;
                  end
                else
                  begin
                    ready_new        = 1'b1;
                    ready_we         = 1'b1;
                    digest_update    = 1'b1;
                    digest_valid_new = 1'b1;
                    digest_valid_we  = 1'b1;
                    sha512_ctrl_new  = CTRL_IDLE;
                    sha512_ctrl_we   = 1'b1;
                  end
              end
            else
              begin
                ready_new        = 1'b1;
                ready_we         = 1'b1;
                digest_update    = 1'b1;
                digest_valid_new = 1'b1;
                digest_valid_we  = 1'b1;
                sha512_ctrl_new  = CTRL_IDLE;
                sha512_ctrl_we   = 1'b1;
              end
          end


        default:
          begin
          end

      endcase // case (sha512_ctrl_reg)
    end // sha512_ctrl_fsm

endmodule // sha512_core


//======================================================================
// kogge_stone_64
// --------------
// High-performance 64-bit Kogge-Stone parallel-prefix adder.
// Reduces carry-propagation depth to O(log N) stages.
//======================================================================
module kogge_stone_64(
    input wire [63:0] a,
    input wire [63:0] b,
    output wire [63:0] sum
);
  wire [63:0] g0, p0;
  assign g0 = a & b;
  assign p0 = a ^ b;

  wire [63:0] g1, p1;
  assign g1[0] = g0[0];
  assign p1[0] = p0[0];
  genvar i1;
  generate
    for (i1 = 1; i1 < 64; i1 = i1 + 1) begin : stage1
      assign g1[i1] = g0[i1] | (p0[i1] & g0[i1-1]);
      assign p1[i1] = p0[i1] & p0[i1-1];
    end
  endgenerate

  wire [63:0] g2, p2;
  assign g2[1:0] = g1[1:0];
  assign p2[1:0] = p1[1:0];
  genvar i2;
  generate
    for (i2 = 2; i2 < 64; i2 = i2 + 1) begin : stage2
      assign g2[i2] = g1[i2] | (p1[i2] & g1[i2-2]);
      assign p2[i2] = p1[i2] & p1[i2-2];
    end
  endgenerate

  wire [63:0] g3, p3;
  assign g3[3:0] = g2[3:0];
  assign p3[3:0] = p2[3:0];
  genvar i3;
  generate
    for (i3 = 4; i3 < 64; i3 = i3 + 1) begin : stage3
      assign g3[i3] = g2[i3] | (p2[i3] & g2[i3-4]);
      assign p3[i3] = p2[i3] & p2[i3-4];
    end
  endgenerate

  wire [63:0] g4, p4;
  assign g4[7:0] = g3[7:0];
  assign p4[7:0] = p3[7:0];
  genvar i4;
  generate
    for (i4 = 8; i4 < 64; i4 = i4 + 1) begin : stage4
      assign g4[i4] = g3[i4] | (p3[i4] & g3[i4-8]);
      assign p4[i4] = p3[i4] & p3[i4-8];
    end
  endgenerate

  wire [63:0] g5, p5;
  assign g5[15:0] = g4[15:0];
  assign p5[15:0] = p4[15:0];
  genvar i5;
  generate
    for (i5 = 16; i5 < 64; i5 = i5 + 1) begin : stage5
      assign g5[i5] = g4[i5] | (p4[i5] & g4[i5-16]);
      assign p5[i5] = p4[i5] & p4[i5-16];
    end
  endgenerate

  wire [63:0] g6, p6;
  assign g6[31:0] = g5[31:0];
  assign p6[31:0] = p5[31:0];
  genvar i6;
  generate
    for (i6 = 32; i6 < 64; i6 = i6 + 1) begin : stage6
      assign g6[i6] = g5[i6] | (p5[i6] & g5[i6-32]);
      assign p6[i6] = p5[i6] & p5[i6-32];
    end
  endgenerate

  assign sum[0] = p0[0];
