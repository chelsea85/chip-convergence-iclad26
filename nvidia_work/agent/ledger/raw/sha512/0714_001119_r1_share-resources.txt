// FILE: sha512_core.v
//======================================================================
//
// sha512_core.v
// -------------
// Verilog 2001 implementation of the SHA-512 hash function.
// This is the internal core with wide interfaces.
//
// Optimized for PPA on ASAP7 7nm library.
// Strategy: share-resources, one-hot FSM, flattened arithmetic.
//
//======================================================================

`default_nettype none

module sha512_core(
                   input wire            clk,
                   input wire            reset_n,

                   input wire            init,
                   input wire            next,
                   input wire [1 : 0]    mode,

                   input wire            work_factor,
                   input wire [31 : 0]   work_factor_num,

                   input wire [1023 : 0] block,

                   output wire           ready,
                   output wire [511 : 0] digest,
                   output wire           digest_valid
                  );

  //----------------------------------------------------------------
  // Internal constant and parameter definitions.
  //----------------------------------------------------------------
  localparam SHA512_ROUNDS = 7'd79;

  localparam STATE_IDLE   = 3'b001;
  localparam STATE_ROUNDS = 3'b010;
  localparam STATE_DONE   = 3'b100;

  //----------------------------------------------------------------
  // Registers including update variables and write enable.
  //----------------------------------------------------------------
  reg [63 : 0] a_reg, b_reg, c_reg, d_reg, e_reg, f_reg, g_reg, h_reg;
  reg [63 : 0] a_new, b_new, c_new, d_new, e_new, f_new, g_new, h_new;
  reg          a_h_we;

  reg [63 : 0] H0_reg, H1_reg, H2_reg, H3_reg, H4_reg, H5_reg, H6_reg, H7_reg;
  reg [63 : 0] H0_new, H1_new, H2_new, H3_new, H4_new, H5_new, H6_new, H7_new;
  reg          H_we;

  reg [6 : 0]  round_ctr_reg;
  reg [31 : 0] work_factor_ctr_reg;
  reg          ready_reg;
  reg          digest_valid_reg;
  reg [2 : 0]  sha512_ctrl_reg;

  //----------------------------------------------------------------
  // Wires.
  //----------------------------------------------------------------
  reg digest_init, digest_update;
  reg state_init, state_update;
  reg first_block;
  reg w_init, w_next;
  reg round_ctr_rst, round_ctr_inc;
  reg work_factor_ctr_rst, work_factor_ctr_inc;
  reg ready_new, ready_we;
  reg digest_valid_new, digest_valid_we;
  reg [2 : 0] sha512_ctrl_new;
  reg sha512_ctrl_we;

  wire [63 : 0] k_data, w_data;
  wire [63 : 0] H0_0, H0_1, H0_2, H0_3, H0_4, H0_5, H0_6, H0_7;

  // Bitwise logic wires [rtlscout-cutpoints]
  wire [63 : 0] sum1 = {e_reg[13 : 0], e_reg[63 : 14]} ^ {e_reg[17 : 0], e_reg[63 : 18]} ^ {e_reg[40 : 0], e_reg[63 : 41]};
  wire [63 : 0] ch   = (e_reg & f_reg) ^ ((~e_reg) & g_reg);
  wire [63 : 0] sum0 = {a_reg[27 : 0], a_reg[63 : 28]} ^ {a_reg[33 : 0], a_reg[63 : 34]} ^ {a_reg[38 : 0], a_reg[63 : 39]};
  wire [63 : 0] maj  = (a_reg & b_reg) ^ (a_reg & c_reg) ^ (b_reg & c_reg);

  // Shared arithmetic resources [share-resources]
  // t1 is a 5-operand sum. a_new is t1 + t2 (7 operands total). e_new is t1 + d_reg (6 operands).
  // Flattening these sums allows ABC to optimize the carry-save tree [p2-pure-sum-noop].
  wire [63 : 0] t1 = h_reg + sum1 + ch + k_data + w_data;
  wire [63 : 0] t2 = sum0 + maj;

  // Mutually exclusive adders for state update and digest update
  wire [63 : 0] adder0_res = state_update ? (t1 + t2) : (H0_reg + a_reg);
  wire [63 : 0] adder4_res = state_update ? (d_reg + t1) : (H4_reg + e_reg);

  // Shared incrementer for counters
  wire [31 : 0] common_inc_in  = state_update ? {25'h0, round_ctr_reg} : work_factor_ctr_reg;
  wire [31 : 0] common_inc_out = common_inc_in + 32'h1;

  //----------------------------------------------------------------
  // Module instantiations.
  //----------------------------------------------------------------
  sha512_k_constants k_constants_inst(.addr(round_ctr_reg), .K(k_data));
  sha512_h_constants h_constants_inst(.mode(mode), .H0(H0_0), .H1(H0_1), .H2(H0_2), .H3(H0_3), .H4(H0_4), .H5(H0_5), .H6(H0_6), .H7(H0_7));
  sha512_w_mem w_mem_inst(.clk(clk), .reset_n(reset_n), .block(block), .init(w_init), .next(w_next), .w(w_data));

  //----------------------------------------------------------------
  // Concurrent connectivity for ports etc.
  //----------------------------------------------------------------
  assign ready = ready_reg;
  assign digest = {H0_reg, H1_reg, H2_reg, H3_reg, H4_reg, H5_reg, H6_reg, H7_reg};
  assign digest_valid = digest_valid_reg;

  //----------------------------------------------------------------
  // reg_update
  //----------------------------------------------------------------
  always @ (posedge clk or negedge reset_n)
    begin : reg_update
      if (!reset_n)
        begin
          {a_reg, b_reg, c_reg, d_reg, e_reg, f_reg, g_reg, h_reg} <= 512'h0;
          {H0_reg, H1_reg, H2_reg, H3_reg, H4_reg, H5_reg, H6_reg, H7_reg} <= 512'h0;
          work_factor_ctr_reg <= 32'h0;
          ready_reg           <= 1'b1;
          digest_valid_reg    <= 1'b0;
          round_ctr_reg       <= 7'h0;
          sha512_ctrl_reg     <= STATE_IDLE;
        end
      else
        begin
          if (a_h_we) begin
            a_reg <= a_new; b_reg <= b_new; c_reg <= c_new; d_reg <= d_new;
            e_reg <= e_new; f_reg <= f_new; g_reg <= g_new; h_reg <= h_new;
          end
          if (H_we) begin
            H0_reg <= H0_new; H1_reg <= H1_new; H2_reg <= H2_new; H3_reg <= H3_new;
            H4_reg <= H4_new; H5_reg <= H5_new; H6_reg <= H6_new; H7_reg <= H7_new;
          end
          if (round_ctr_rst) round_ctr_reg <= 7'h0;
          else if (round_ctr_inc) round_ctr_reg <= common_inc_out[6 : 0];
          if (work_factor_ctr_rst) work_factor_ctr_reg <= 32'h0;
          else if (work_factor_ctr_inc) work_factor_ctr_reg <= common_inc_out;
          if (ready_we) ready_reg <= ready_new;
          if (digest_valid_we) digest_valid_reg <= digest_valid_new;
          if (sha512_ctrl_we) sha512_ctrl_reg <= sha512_ctrl_new;
        end
    end

  //----------------------------------------------------------------
  // digest_logic
  //----------------------------------------------------------------
  always @*
    begin : digest_logic
      H0_new = H0_reg; H1_new = H1_reg; H2_new = H2_reg; H3_new = H3_reg;
      H4_new = H4_reg; H5_new = H5_reg; H6_new = H6_reg; H7_new = H7_reg;
      H_we = 1'b0;
      if (digest_init) begin
        H0_new = H0_0; H1_new = H0_1; H2_new = H0_2; H3_new = H0_3;
        H4_new = H0_4; H5_new = H0_5; H6_new = H0_6; H7_new = H0_7;
        H_we = 1'b1;
      end else if (digest_update) begin
        H0_new = adder0_res;
        H1_new = H1_reg + b_reg;
        H2_new = H2_reg + c_reg;
        H3_new = H3_reg + d_reg;
        H4_new = adder4_res;
        H5_new = H5_reg + f_reg;
        H6_new = H6_reg + g_reg;
        H7_new = H7_reg + h_reg;
        H_we = 1'b1;
      end
    end

  //----------------------------------------------------------------
  // state_logic
  //----------------------------------------------------------------
  always @*
    begin : state_logic
      a_new = a_reg; b_new = b_reg; c_new = c_reg; d_new = d_reg;
      e_new = e_reg; f_new = f_reg; g_new = g_reg; h_new = h_reg;
      a_h_we = 1'b0;
      if (state_init) begin
        a_new = first_block ? H0_0 : H0_reg;
        b_new = first_block ? H0_1 : H1_reg;
        c_new = first_block ? H0_2 : H2_reg;
        d_new = first_block ? H0_3 : H3_reg;
        e_new = first_block ? H0_4 : H4_reg;
        f_new = first_block ? H0_5 : H5_reg;
        g_new = first_block ? H0_6 : H6_reg;
        h_new = first_block ? H0_7 : H7_reg;
        a_h_we = 1'b1;
      end else if (state_update) begin
        a_new = adder0_res;
        b_new = a_reg;
        c_new = b_reg;
        d_new = c_reg;
        e_new = adder4_res;
        f_new = e_reg;
        g_new = f_reg;
        h_new = g_reg;
        a_h_we = 1'b1;
      end
    end

  //----------------------------------------------------------------
  // sha512_ctrl_fsm
  //----------------------------------------------------------------
  always @*
    begin : sha512_ctrl_fsm
      digest_init = 1'b0; digest_update = 1'b0; state_init = 1'b0; state_update = 1'b0;
      first_block = 1'b0; w_init = 1'b0; w_next = 1'b0; round_ctr_inc = 1'b0;
      round_ctr_rst = 1'b0; digest_valid_new = 1'b0; digest_valid_we = 1'b0;
      work_factor_ctr_rst = 1'b0; work_factor_ctr_inc = 1'b0; ready_new = 1'b0;
      ready_we = 1'b0; sha512_ctrl_new = STATE_IDLE; sha512_ctrl_we = 1'b0;

      case (sha512_ctrl_reg)
        STATE_IDLE: begin
          if (init || next) begin
            ready_new = 1'b0; ready_we = 1'b1;
            work_factor_ctr_rst = 1'b1;
            w_init = 1'b1; state_init = 1'b1; round_ctr_rst = 1'b1;
            digest_valid_new = 1'b0; digest_valid_we = 1'b1;
            sha512_ctrl_new = STATE_ROUNDS; sha512_ctrl_we = 1'b1;
            if (init) begin digest_init = 1'b1; first_block = 1'b1; end
          end
        end

        STATE_ROUNDS: begin
          w_next = 1'b1; state_update = 1'b1; round_ctr_inc = 1'b1;
          if (round_ctr_reg == SHA512_ROUNDS) begin
            work_factor_ctr_inc = 1'b1;
            sha512_ctrl_new = STATE_DONE; sha512_ctrl_we = 1'b1;
          end
        end

        STATE_DONE: begin
          if (work_factor && (work_factor_ctr_reg < work_factor_num)) begin
            w_init = 1'b1; state_init = 1'b1; round_ctr_rst = 1'b1;
            sha512_ctrl_new = STATE_ROUNDS; sha512_ctrl_we = 1'b1;
          end else begin
            ready_new = 1'b1; ready_we = 1'b1;
            digest_update = 1'b1; digest_valid_new = 1'b1; digest_valid_we = 1'b1;
            sha512_ctrl_new = STATE_IDLE; sha512_ctrl_we = 1'b1;
          end
        end
        default: sha512_ctrl_new = STATE_IDLE;
      endcase
    end
endmodule

// FILE: sha512_w_mem.v
//======================================================================
//
// sha512_w_mem.v
// --------------
// The W memory for the SHA-512 core.
//
// Optimized for PPA on ASAP7 7nm library.
// Strategy: flattened arithmetic, simplified muxing.
//
//======================================================================

`default_nettype none

module sha512_w_mem(
                    input wire            clk,
                    input wire            reset_n,
                    input wire [1023 : 0] block,
                    input wire            init,
                    input wire            next,
                    output wire [63 : 0]  w
                   );

  reg [63 : 0] w_mem [0 : 15];
  reg [6 : 0]  w_ctr_reg;

  // Flattened w_new logic [p2-pure-sum-noop]
  wire [63 : 0] w_0  = w_mem[0];
  wire [63 : 0] w_1  = w_mem[1];
  wire [63 : 0] w_9  = w_mem[9];
  wire [63 : 0] w_14 = w_mem[14];

  wire [63 : 0] d0 = {w_1[0],     w_1[63 : 1]} ^ {w_1[7 : 0], w_1[63 : 8]} ^ {7'b0, w_1[63 : 7]};
  wire [63 : 0] d1 = {w_14[18 : 0], w_14[63 : 19]} ^ {w_14[60 : 0], w_14[63 : 61]} ^ {6'b0, w_14[63 : 6]};

  wire [63 : 0] w_new = w_0 + d0 + w_9 + d1;

  // Simplified w_tmp selection [rtlscout-cutpoints]
  // w_ctr_reg < 16 is equivalent to w_ctr_reg[6:4] == 0
  assign w = (w_ctr_reg[6 : 4] == 3'b000) ? w_mem[w_ctr_reg[3 : 0]] : w_new;

  always @ (posedge clk or negedge reset_n)
    begin : reg_update
      integer i;
      if (!reset_n)
        begin
          for (i = 0; i < 16; i = i + 1) w_mem[i] <= 64'h0;
          w_ctr_reg <= 7'h0;
        end
      else
        begin
          if (init)
            begin
              w_mem[00] <= block[1023 : 960]; w_mem[01] <= block[959  : 896];
              w_mem[02] <= block[895  : 832]; w_mem[03] <= block[831  : 768];
              w_mem[04] <= block[767  : 704]; w_mem[05] <= block[703  : 640];
              w_mem[06] <= block[639  : 576]; w_mem[07] <= block[575  : 512];
              w_mem[08] <= block[511  : 448]; w_mem[09] <= block[447  : 384];
              w_mem[10] <= block[383  : 320]; w_mem[11] <= block[319  : 256];
              w_mem[12] <= block[255  : 192]; w_mem[13] <= block[191  : 128];
              w_mem[14] <= block[127  :  64]; w_mem[15] <= block[63   :   0];
              w_ctr_reg <= 7'h00;
            end
          else if (next)
            begin
              w_ctr_reg <= w_ctr_reg + 7'h01;
              if (w_ctr_reg > 7'd15)
                begin
                  for (i = 0; i < 15; i = i + 1) w_mem[i] <= w_mem[i+1];
                  w_mem[15] <= w_new;
                end
            end
        end
    end
endmodule
