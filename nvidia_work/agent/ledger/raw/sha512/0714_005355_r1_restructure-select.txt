// FILE: sha512_core.v
//======================================================================
//
// sha512_core.v
// -------------
// Verilog 2001 implementation of the SHA-512 hash function.
// This is the internal core with wide interfaces.
//
// Optimized for PPA on ASAP7 7nm.
// Strategy: Balanced addition trees for the 7-operand round logic
// to reduce carry-chain depth (WNS bottleneck).
//
//======================================================================

`default_nettype none

module sha512_core(
                   input wire            clk,
                   input wire            reset_n,

                   input wire            init,
                   input wire            next,
                   input wire [1 : 0]    mode,

                   input wire            work_factor,
                   input wire [31 : 0]   work_factor_num,

                   input wire [1023 : 0] block,

                   output wire           ready,
                   output wire [511 : 0] digest,
                   output wire           digest_valid
                  );


  //----------------------------------------------------------------
  // Internal constant and parameter definitions.
  //----------------------------------------------------------------
  localparam SHA512_ROUNDS = 79;

  localparam CTRL_IDLE   = 2'h0;
  localparam CTRL_ROUNDS = 2'h1;
  localparam CTRL_DONE   = 2'h2;


  //----------------------------------------------------------------
  // Registers including update variables and write enable.
  //----------------------------------------------------------------
  reg [63 : 0] a_reg;
  reg [63 : 0] a_new;
  reg [63 : 0] b_reg;
  reg [63 : 0] b_new;
  reg [63 : 0] c_reg;
  reg [63 : 0] c_new;
  reg [63 : 0] d_reg;
  reg [63 : 0] d_new;
  reg [63 : 0] e_reg;
  reg [63 : 0] e_new;
  reg [63 : 0] f_reg;
  reg [63 : 0] f_new;
  reg [63 : 0] g_reg;
  reg [63 : 0] g_new;
  reg [63 : 0] h_reg;
  reg [63 : 0] h_new;
  reg          a_h_we;

  reg [63 : 0] H0_reg;
  reg [63 : 0] H0_new;
  reg [63 : 0] H1_reg;
  reg [63 : 0] H1_new;
  reg [63 : 0] H2_reg;
  reg [63 : 0] H2_new;
  reg [63 : 0] H3_reg;
  reg [63 : 0] H3_new;
  reg [63 : 0] H4_reg;
  reg [63 : 0] H4_new;
  reg [63 : 0] H5_reg;
  reg [63 : 0] H5_new;
  reg [63 : 0] H6_reg;
  reg [63 : 0] H6_new;
  reg [63 : 0] H7_reg;
  reg [63 : 0] H7_new;
  reg          H_we;

  reg [6 : 0]  round_ctr_reg;
  reg [6 : 0]  round_ctr_new;
  reg          round_ctr_we;
  reg          round_ctr_inc;
  reg          round_ctr_rst;

  reg [31 : 0] work_factor_ctr_reg;
  reg [31 : 0] work_factor_ctr_new;
  reg          work_factor_ctr_rst;
  reg          work_factor_ctr_inc;
  reg          work_factor_ctr_we;

  reg          ready_reg;
  reg          ready_new;
  reg          ready_we;

  reg          digest_valid_reg;
  reg          digest_valid_new;
  reg          digest_valid_we;

  reg [1 : 0]  sha512_ctrl_reg;
  reg [1 : 0]  sha512_ctrl_new;
  reg          sha512_ctrl_we;


  //----------------------------------------------------------------
  // Wires.
  //----------------------------------------------------------------
  reg digest_init;
  reg digest_update;

  reg state_init;
  reg state_update;

  reg first_block;

  wire [63 : 0] t1;
  wire [63 : 0] t2;

  wire [63 : 0] k_data;

  reg           w_init;
  reg           w_next;
  wire [63 : 0] w_data;

  wire [63 : 0] H0_0;
  wire [63 : 0] H0_1;
  wire [63 : 0] H0_2;
  wire [63 : 0] H0_3;
  wire [63 : 0] H0_4;
  wire [63 : 0] H0_5;
  wire [63 : 0] H0_6;
  wire [63 : 0] H0_7;


  //----------------------------------------------------------------
  // Module instantiantions.
  //----------------------------------------------------------------
  sha512_k_constants k_constants_inst(
                                      .addr(round_ctr_reg),
                                      .K(k_data)
                                     );


  sha512_h_constants h_constants_inst(
                                      .mode(mode),

                                      .H0(H0_0),
                                      .H1(H0_1),
                                      .H2(H0_2),
                                      .H3(H0_3),
                                      .H4(H0_4),
                                      .H5(H0_5),
                                      .H6(H0_6),
                                      .H7(H0_7)
                                     );


  sha512_w_mem w_mem_inst(
                          .clk(clk),
                          .reset_n(reset_n),

                          .block(block),

                          .init(w_init),
                          .next(w_next),
                          .w(w_data)
                         );


  //----------------------------------------------------------------
  // Concurrent connectivity for ports etc.
  //----------------------------------------------------------------
  assign ready = ready_reg;

  assign digest = {H0_reg, H1_reg, H2_reg, H3_reg,
                   H4_reg, H5_reg, H6_reg, H7_reg};

  assign digest_valid = digest_valid_reg;


  //----------------------------------------------------------------
  // reg_update
  //----------------------------------------------------------------
  always @ (posedge clk or negedge reset_n)
    begin : reg_update
      if (!reset_n)
        begin
          a_reg               <= 64'h0;
          b_reg               <= 64'h0;
          c_reg               <= 64'h0;
          d_reg               <= 64'h0;
          e_reg               <= 64'h0;
          f_reg               <= 64'h0;
          g_reg               <= 64'h0;
          h_reg               <= 64'h0;
          H0_reg              <= 64'h0;
          H1_reg              <= 64'h0;
          H2_reg              <= 64'h0;
          H3_reg              <= 64'h0;
          H4_reg              <= 64'h0;
          H5_reg              <= 64'h0;
          H6_reg              <= 64'h0;
          H7_reg              <= 64'h0;
          work_factor_ctr_reg <= 32'h0;
          ready_reg           <= 1'b1;
          digest_valid_reg    <= 1'b0;
          round_ctr_reg       <= 7'h0;
          sha512_ctrl_reg     <= CTRL_IDLE;
        end

      else
        begin
          if (a_h_we)
            begin
              a_reg <= a_new;
              b_reg <= b_new;
              c_reg <= c_new;
              d_reg <= d_new;
              e_reg <= e_new;
              f_reg <= f_new;
              g_reg <= g_new;
              h_reg <= h_new;
            end

          if (H_we)
            begin
              H0_reg <= H0_new;
              H1_reg <= H1_new;
              H2_reg <= H2_new;
              H3_reg <= H3_new;
              H4_reg <= H4_new;
              H5_reg <= H5_new;
              H6_reg <= H6_new;
              H7_reg <= H7_new;
            end

          if (round_ctr_we)
            round_ctr_reg <= round_ctr_new;

          if (work_factor_ctr_we)
            work_factor_ctr_reg <= work_factor_ctr_new;

          if (ready_we)
            ready_reg <= ready_new;

          if (digest_valid_we)
            digest_valid_reg <= digest_valid_new;

          if (sha512_ctrl_we)
            sha512_ctrl_reg <= sha512_ctrl_new;
        end
    end // reg_update


  //----------------------------------------------------------------
  // digest_logic
  //----------------------------------------------------------------
  always @*
    begin : digest_logic
      H0_new = H0_reg;
      H1_new = H1_reg;
      H2_new = H2_reg;
      H3_new = H3_reg;
      H4_new = H4_reg;
      H5_new = H5_reg;
      H6_new = H6_reg;
      H7_new = H7_reg;
      H_we = 1'b0;

      if (digest_init)
        begin
          H0_new = H0_0;
          H1_new = H0_1;
          H2_new = H0_2;
          H3_new = H0_3;
          H4_new = H0_4;
          H5_new = H0_5;
          H6_new = H0_6;
          H7_new = H0_7;
          H_we = 1'b1;
        end
      else if (digest_update)
        begin
          H0_new = H0_reg + a_reg;
          H1_new = H1_reg + b_reg;
          H2_new = H2_reg + c_reg;
          H3_new = H3_reg + d_reg;
          H4_new = H4_reg + e_reg;
          H5_new = H5_reg + f_reg;
          H6_new = H6_reg + g_reg;
          H7_new = H7_reg + h_reg;
          H_we = 1'b1;
        end
    end // digest_logic


  //----------------------------------------------------------------
  // Compression Function Logic (Balanced Tree)
  //----------------------------------------------------------------
  wire [63 : 0] sum1 = {e_reg[13 : 0], e_reg[63 : 14]} ^
                       {e_reg[17 : 0], e_reg[63 : 18]} ^
                       {e_reg[40 : 0], e_reg[63 : 41]};

  wire [63 : 0] ch   = (e_reg & f_reg) ^ ((~e_reg) & g_reg);

  wire [63 : 0] sum0 = {a_reg[27 : 0], a_reg[63 : 28]} ^
                       {a_reg[33 : 0], a_reg[63 : 34]} ^
                       {a_reg[38 : 0], a_reg[63 : 39]};

  wire [63 : 0] maj  = (a_reg & b_reg) ^ (a_reg & c_reg) ^ (b_reg & c_reg);

  // Balanced Addition Tree for SHA-512 Round
  // a_new = h + sum1 + ch + k + w + sum0 + maj
  // e_new = d + h + sum1 + ch + k + w
  
  wire [63 : 0] s1_ch = sum1 + ch;
  wire [63 : 0] k_w   = k_data + w_data;
  wire [63 : 0] s0_m  = sum0 + maj;
  
  wire [63 : 0] common_sum = s1_ch + k_w; // (sum1+ch) + (k+w)
  wire [63 : 0] h_s0_m     = h_reg + s0_m;
  wire [63 : 0] h_d        = h_reg + d_reg;

  assign t1 = common_sum + h_reg; // Original t1 definition
  assign t2 = s0_m;               // Original t2 definition

  wire [63 : 0] a_new_round = common_sum + h_s0_m; // Balanced 7-operand sum
  wire [63 : 0] e_new_round = common_sum + h_d;    // Balanced 6-operand sum


  //----------------------------------------------------------------
  // state_logic
  //----------------------------------------------------------------
  always @*
    begin : state_logic
      a_new  = a_reg;
      b_new  = b_reg;
      c_new  = c_reg;
      d_new  = d_reg;
      e_new  = e_reg;
      f_new  = f_reg;
      g_new  = g_reg;
      h_new  = h_reg;
      a_h_we = 1'b0;

      if (state_init)
        begin
          a_h_we = 1'b1;
          if (first_block)
            begin
              a_new  = H0_0;
              b_new  = H0_1;
              c_new  = H0_2;
              d_new  = H0_3;
              e_new  = H0_4;
              f_new  = H0_5;
              g_new  = H0_6;
              h_new  = H0_7;
            end
          else
            begin
              a_new  = H0_reg;
              b_new  = H1_reg;
              c_new  = H2_reg;
              d_new  = H3_reg;
              e_new  = H4_reg;
              f_new  = H5_reg;
              g_new  = H6_reg;
              h_new  = H7_reg;
            end
        end
      else if (state_update)
        begin
          a_h_we = 1'b1;
          a_new  = a_new_round;
          b_new  = a_reg;
          c_new  = b_reg;
          d_new  = c_reg;
          e_new  = e_new_round;
          f_new  = e_reg;
          g_new  = f_reg;
          h_new  = g_reg;
        end
    end // state_logic


  //----------------------------------------------------------------
  // round_ctr
  //----------------------------------------------------------------
  always @*
    begin : round_ctr
      round_ctr_new = 7'h00;
      round_ctr_we  = 1'b0;

      if (round_ctr_rst)
        begin
          round_ctr_new = 7'h00;
          round_ctr_we  = 1'b1;
        end
      else if (round_ctr_inc)
        begin
          round_ctr_new = round_ctr_reg + 1'b1;
          round_ctr_we  = 1'b1;
        end
    end // round_ctr


  //----------------------------------------------------------------
  // work_factor_ctr
  //----------------------------------------------------------------
  always @*
    begin : work_factor_ctr
      work_factor_ctr_new  = 32'h0;
      work_factor_ctr_we   = 1'b0;

      if (work_factor_ctr_rst)
        begin
          work_factor_ctr_new = 32'h0;
          work_factor_ctr_we  = 1'b1;
        end
      else if (work_factor_ctr_inc)
        begin
          work_factor_ctr_new = work_factor_ctr_reg + 1'b1;
          work_factor_ctr_we  = 1'b1;
        end
    end // work_factor_ctr


  //----------------------------------------------------------------
  // sha512_ctrl_fsm
  //----------------------------------------------------------------
  always @*
    begin : sha512_ctrl_fsm
      digest_init         = 1'b0;
      digest_update       = 1'b0;
      state_init          = 1'b0;
      state_update        = 1'b0;
      first_block         = 1'b0;
      w_init              = 1'b0;
      w_next              = 1'b0;
      round_ctr_inc       = 1'b0;
      round_ctr_rst       = 1'b0;
      digest_valid_new    = 1'b0;
      digest_valid_we     = 1'b0;
      work_factor_ctr_rst = 1'b0;
      work_factor_ctr_inc = 1'b0;
      ready_new           = 1'b0;
      ready_we            = 1'b0;
      sha512_ctrl_new     = CTRL_IDLE;
      sha512_ctrl_we      = 1'b0;

      case (sha512_ctrl_reg)
        CTRL_IDLE:
          begin
            if (init)
              begin
                ready_new           = 1'b0;
                ready_we            = 1'b1;
                work_factor_ctr_rst = 1'b1;
                digest_init         = 1'b1;
                w_init              = 1'b1;
                state_init          = 1'b1;
                first_block         = 1'b1;
                round_ctr_rst       = 1'b1;
                digest_valid_new    = 1'b0;
                digest_valid_we     = 1'b1;
                sha512_ctrl_new     = CTRL_ROUNDS;
                sha512_ctrl_we      = 1'b1;
              end
            else if (next)
              begin
                ready_new           = 1'b0;
                ready_we            = 1'b1;
                work_factor_ctr_rst = 1'b1;
                w_init              = 1'b1;
                state_init          = 1'b1;
                round_ctr_rst       = 1'b1;
                digest_valid_new    = 1'b0;
                digest_valid_we     = 1'b1;
                sha512_ctrl_new     = CTRL_ROUNDS;
                sha512_ctrl_we      = 1'b1;
              end
          end


        CTRL_ROUNDS:
          begin
            w_next        = 1'b1;
            state_update  = 1'b1;
            round_ctr_inc = 1'b1;

            if (round_ctr_reg == SHA512_ROUNDS)
              begin
                work_factor_ctr_inc = 1'b1;
                sha512_ctrl_new     = CTRL_DONE;
                sha512_ctrl_we      = 1'b1;
              end
          end


        CTRL_DONE:
          begin
            if (work_factor)
              begin
                if (work_factor_ctr_reg < work_factor_num)
                  begin
                    w_init              = 1'b1;
                    state_init          = 1'b1;
                    round_ctr_rst       = 1'b1;
                    sha512_ctrl_new     = CTRL_ROUNDS;
                    sha512_ctrl_we      = 1'b1;
                  end
                else
                  begin
                    ready_new        = 1'b1;
                    ready_we         = 1'b1;
                    digest_update    = 1'b1;
                    digest_valid_new = 1'b1;
                    digest_valid_we  = 1'b1;
                    sha512_ctrl_new  = CTRL_IDLE;
                    sha512_ctrl_we   = 1'b1;
                  end
              end
            else
              begin
                ready_new        = 1'b1;
                ready_we         = 1'b1;
                digest_update    = 1'b1;
                digest_valid_new = 1'b1;
                digest_valid_we  = 1'b1;
                sha512_ctrl_new  = CTRL_IDLE;
                sha512_ctrl_we   = 1'b1;
              end
          end

        default: ;
      endcase
    end // sha512_ctrl_fsm

endmodule // sha512_core
