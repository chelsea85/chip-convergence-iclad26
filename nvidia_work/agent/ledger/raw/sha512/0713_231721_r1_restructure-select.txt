// FILE: sha512_core.v
//======================================================================
//
// sha512_core.v
// -------------
// Verilog 2001 implementation of the SHA-512 hash function.
// This is the internal core with wide interfaces.
//
// Optimized for ASAP7 7nm library with a focus on timing (WNS).
// Strategy: Restructure arithmetic chains and selection logic.
//
//======================================================================

`default_nettype none

module sha512_core(
                   input wire            clk,
                   input wire            reset_n,

                   input wire            init,
                   input wire            next,
                   input wire [1 : 0]    mode,

                   input wire            work_factor,
                   input wire [31 : 0]   work_factor_num,

                   input wire [1023 : 0] block,

                   output wire           ready,
                   output wire [511 : 0] digest,
                   output wire           digest_valid
                  );

  //----------------------------------------------------------------
  // Internal constant and parameter definitions.
  //----------------------------------------------------------------
  localparam SHA512_ROUNDS = 79;

  localparam CTRL_IDLE   = 2'h0;
  localparam CTRL_ROUNDS = 2'h1;
  localparam CTRL_DONE   = 2'h2;

  //----------------------------------------------------------------
  // Registers.
  //----------------------------------------------------------------
  reg [63 : 0] a_reg, b_reg, c_reg, d_reg, e_reg, f_reg, g_reg, h_reg;
  reg [63 : 0] a_new, b_new, c_new, d_new, e_new, f_new, g_new, h_new;
  reg          a_h_we;

  reg [63 : 0] H0_reg, H1_reg, H2_reg, H3_reg, H4_reg, H5_reg, H6_reg, H7_reg;
  reg [63 : 0] H0_new, H1_new, H2_new, H3_new, H4_new, H5_new, H6_new, H7_new;
  reg          H_we;

  reg [6 : 0]  round_ctr_reg;
  reg [6 : 0]  round_ctr_new;
  reg          round_ctr_we;
  reg          round_ctr_inc;
  reg          round_ctr_rst;

  reg [31 : 0] work_factor_ctr_reg;
  reg [31 : 0] work_factor_ctr_new;
  reg          work_factor_ctr_rst;
  reg          work_factor_ctr_inc;
  reg          work_factor_ctr_we;

  reg          ready_reg;
  reg          ready_new;
  reg          ready_we;

  reg          digest_valid_reg;
  reg          digest_valid_new;
  reg          digest_valid_we;

  reg [1 : 0]  sha512_ctrl_reg;
  reg [1 : 0]  sha512_ctrl_new;
  reg          sha512_ctrl_we;

  //----------------------------------------------------------------
  // Wires.
  //----------------------------------------------------------------
  reg digest_init;
  reg digest_update;
  reg state_init;
  reg state_update;
  reg first_block;

  wire [63 : 0] k_data;
  reg           w_init;
  reg           w_next;
  wire [63 : 0] w_data;

  wire [63 : 0] H0_0, H0_1, H0_2, H0_3, H0_4, H0_5, H0_6, H0_7;

  //----------------------------------------------------------------
  // Module instantiations.
  //----------------------------------------------------------------
  sha512_k_constants k_constants_inst(.addr(round_ctr_reg), .K(k_data));
  sha512_h_constants h_constants_inst(.mode(mode), .H0(H0_0), .H1(H0_1), .H2(H0_2), .H3(H0_3), .H4(H0_4), .H5(H0_5), .H6(H0_6), .H7(H0_7));
  sha512_w_mem w_mem_inst(.clk(clk), .reset_n(reset_n), .block(block), .init(w_init), .next(w_next), .w(w_data));

  //----------------------------------------------------------------
  // Concurrent connectivity.
  //----------------------------------------------------------------
  assign ready = ready_reg;
  assign digest = {H0_reg, H1_reg, H2_reg, H3_reg, H4_reg, H5_reg, H6_reg, H7_reg};
  assign digest_valid = digest_valid_reg;

  //----------------------------------------------------------------
  // Arithmetic Logic (Balanced Trees)
  //----------------------------------------------------------------
  wire [63 : 0] s0 = {a_reg[27:0], a_reg[63:28]} ^ {a_reg[33:0], a_reg[63:34]} ^ {a_reg[38:0], a_reg[63:39]};
  wire [63 : 0] s1 = {e_reg[13:0], e_reg[63:14]} ^ {e_reg[17:0], e_reg[63:18]} ^ {e_reg[40:0], e_reg[63:41]};
  wire [63 : 0] ch = (e_reg & f_reg) ^ ((~e_reg) & g_reg);
  wire [63 : 0] maj = (a_reg & b_reg) ^ (a_reg & c_reg) ^ (b_reg & c_reg);

  // Balanced 5-operand sum for T1: t1 = h + s1 + ch + k + w
  // Balanced 2-operand sum for T2: t2 = s0 + maj
  // a_new = t1 + t2 (7 operands total)
  // e_new = d + t1 (6 operands total)
  wire [63 : 0] t1_sum_h_s1 = h_reg + s1;
  wire [63 : 0] t1_sum_ch_k = ch + k_data;
  wire [63 : 0] t1_lp       = t1_sum_h_s1 + t1_sum_ch_k;
  wire [63 : 0] t1_final    = t1_lp + w_data;
  wire [63 : 0] t2_final    = s0 + maj;

  //----------------------------------------------------------------
  // Register Update Logic.
  //----------------------------------------------------------------
  always @ (posedge clk or negedge reset_n) begin : reg_update
    if (!reset_n) begin
      a_reg <= 64'h0; b_reg <= 64'h0; c_reg <= 64'h0; d_reg <= 64'h0;
      e_reg <= 64'h0; f_reg <= 64'h0; g_reg <= 64'h0; h_reg <= 64'h0;
      H0_reg <= 64'h0; H1_reg <= 64'h0; H2_reg <= 64'h0; H3_reg <= 64'h0;
      H4_reg <= 64'h0; H5_reg <= 64'h0; H6_reg <= 64'h0; H7_reg <= 64'h0;
      work_factor_ctr_reg <= 32'h0; ready_reg <= 1'b1; digest_valid_reg <= 1'b0;
      round_ctr_reg <= 7'h0; sha512_ctrl_reg <= CTRL_IDLE;
    end else begin
      if (a_h_we) begin
        a_reg <= a_new; b_reg <= b_new; c_reg <= c_new; d_reg <= d_new;
        e_reg <= e_new; f_reg <= f_new; g_reg <= g_new; h_reg <= h_new;
      end
      if (H_we) begin
        H0_reg <= H0_new; H1_reg <= H1_new; H2_reg <= H2_new; H3_reg <= H3_new;
        H4_reg <= H4_new; H5_reg <= H5_new; H6_reg <= H6_new; H7_reg <= H7_new;
      end
      if (round_ctr_we)       round_ctr_reg <= round_ctr_new;
      if (work_factor_ctr_we)  work_factor_ctr_reg <= work_factor_ctr_new;
      if (ready_we)           ready_reg <= ready_new;
      if (digest_valid_we)    digest_valid_reg <= digest_valid_new;
      if (sha512_ctrl_we)     sha512_ctrl_reg <= sha512_ctrl_new;
    end
  end

  //----------------------------------------------------------------
  // Digest and State Logic (Restructured Muxes).
  //----------------------------------------------------------------
  always @* begin : digest_logic
    H_we = digest_init | digest_update;
    H0_new = digest_init ? H0_0 : (H0_reg + a_reg);
    H1_new = digest_init ? H0_1 : (H1_reg + b_reg);
    H2_new = digest_init ? H0_2 : (H2_reg + c_reg);
    H3_new = digest_init ? H0_3 : (H3_reg + d_reg);
    H4_new = digest_init ? H0_4 : (H4_reg + e_reg);
    H5_new = digest_init ? H0_5 : (H5_reg + f_reg);
    H6_new = digest_init ? H0_6 : (H6_reg + g_reg);
    H7_new = digest_init ? H0_7 : (H7_reg + h_reg);
  end

  always @* begin : state_logic
    a_new = 64'h0; b_new = 64'h0; c_new = 64'h0; d_new = 64'h0;
    e_new = 64'h0; f_new = 64'h0; g_new = 64'h0; h_new = 64'h0;
    a_h_we = state_init | state_update;

    if (state_update) begin
      a_new = t1_final + t2_final;
      b_new = a_reg;
      c_new = b_reg;
      d_new = c_reg;
      e_new = d_reg + t1_final;
      f_new = e_reg;
      g_new = f_reg;
      h_new = g_reg;
    end else if (state_init) begin
      a_new = first_block ? H0_0 : H0_reg;
      b_new = first_block ? H0_1 : H1_reg;
      c_new = first_block ? H0_2 : H2_reg;
      d_new = first_block ? H0_3 : H3_reg;
      e_new = first_block ? H0_4 : H4_reg;
      f_new = first_block ? H0_5 : H5_reg;
      g_new = first_block ? H0_6 : H6_reg;
      h_new = first_block ? H0_7 : H7_reg;
    end
  end

  //----------------------------------------------------------------
  // Control Logic.
  //----------------------------------------------------------------
  always @* begin : round_ctr_logic
    round_ctr_new = 7'h00;
    round_ctr_we  = round_ctr_rst | round_ctr_inc;
    if (round_ctr_inc) round_ctr_new = round_ctr_reg + 7'h01;
  end

  always @* begin : work_factor_ctr_logic
    work_factor_ctr_new = 32'h0;
    work_factor_ctr_we  = work_factor_ctr_rst | work_factor_ctr_inc;
    if (work_factor_ctr_inc) work_factor_ctr_new = work_factor_ctr_reg + 32'h1;
  end

  always @* begin : sha512_ctrl_fsm
    digest_init = 0; digest_update = 0; state_init = 0; state_update = 0;
    first_block = 0; w_init = 0; w_next = 0; round_ctr_inc = 0; round_ctr_rst = 0;
    digest_valid_new = 0; digest_valid_we = 0; work_factor_ctr_rst = 0;
    work_factor_ctr_inc = 0; ready_new = 0; ready_we = 0;
    sha512_ctrl_new = CTRL_IDLE; sha512_ctrl_we = 0;

    case (sha512_ctrl_reg)
      CTRL_IDLE: begin
        if (init || next) begin
          ready_new = 0; ready_we = 1; work_factor_ctr_rst = 1;
          w_init = 1; state_init = 1; round_ctr_rst = 1;
          digest_valid_new = 0; digest_valid_we = 1;
          sha512_ctrl_new = CTRL_ROUNDS; sha512_ctrl_we = 1;
          if (init) begin digest_init = 1; first_block = 1; end
        end
      end

      CTRL_ROUNDS: begin
        w_next = 1; state_update = 1; round_ctr_inc = 1;
        if (round_ctr_reg == SHA512_ROUNDS) begin
          work_factor_ctr_inc = 1;
          sha512_ctrl_new = CTRL_DONE; sha512_ctrl_we = 1;
        end
      end

      CTRL_DONE: begin
        if (work_factor && (work_factor_ctr_reg < work_factor_num)) begin
          w_init = 1; state_init = 1; round_ctr_rst = 1;
          sha512_ctrl_new = CTRL_ROUNDS; sha512_ctrl_we = 1;
        end else begin
          ready_new = 1; ready_we = 1; digest_update = 1;
          digest_valid_new = 1; digest_valid_we = 1;
          sha512_ctrl_new = CTRL_IDLE; sha512_ctrl_we = 1;
        end
      end
      default: sha512_ctrl_new = CTRL_IDLE;
    endcase
  end
endmodule

// FILE: sha512_w_mem.v
//======================================================================
//
// sha512_w_mem.v
// --------------
// The W memory for the SHA-512 core.
// Optimized for ASAP7 7nm library.
//
//======================================================================

`default_nettype none

module sha512_w_mem(
                    input wire            clk,
                    input wire            reset_n,
                    input wire [1023 : 0] block,
                    input wire            init,
                    input wire            next,
                    output wire [63 : 0]  w
                   );

  reg [63 : 0] w_mem [0 : 15];
  reg [6 : 0]  w_ctr_reg;
  reg [6 : 0]  w_ctr_new;
  reg          w_ctr_we;

  wire [63 : 0] w_0  = w_mem[0];
  wire [63 : 0] w_1  = w_mem[1];
  wire [63 : 0] w_9  = w_mem[9];
  wire [63 : 0] w_14 = w_mem[14];

  wire [63 : 0] d0 = {w_1[0],     w_1[63 : 1]} ^ {w_1[7 : 0], w_1[63 : 8]} ^ {7'b0, w_1[63 : 7]};
  wire [63 : 0] d1 = {w_14[18 : 0], w_14[63 : 19]} ^ {w_14[60 : 0], w_14[63 : 61]} ^ {6'b0, w_14[63 : 6]};

  // Balanced 4-operand sum: w_new = (w_0 + d0) + (w_9 + d1)
  wire [63 : 0] w_new_a = w_0 + d0;
  wire [63 : 0] w_new_b = w_9 + d1;
  wire [63 : 0] w_new   = w_new_a + w_new_b;

  assign w = (w_ctr_reg < 16) ? w_mem[w_ctr_reg[3 : 0]] : w_new;

  always @ (posedge clk or negedge reset_n) begin : reg_update
    integer i;
    if (!reset_n) begin
      for (i = 0; i < 16; i = i + 1) w_mem[i] <= 64'h0;
      w_ctr_reg <= 7'h0;
    end else begin
      if (init) begin
        w_mem[00] <= block[1023 : 960]; w_mem[01] <= block[959  : 896];
        w_mem[02] <= block[895  : 832]; w_mem[03] <= block[831  : 768];
        w_mem[04] <= block[767  : 704]; w_mem[05] <= block[703  : 640];
        w_mem[06] <= block[639  : 576]; w_mem[07] <= block[575  : 512];
        w_mem[08] <= block[511  : 448]; w_mem[09] <= block[447  : 384];
        w_mem[10] <= block[383  : 320]; w_mem[11] <= block[319  : 256];
        w_mem[12] <= block[255  : 192]; w_mem[13] <= block[191  : 128];
        w_mem[14] <= block[127  :  64]; w_mem[15] <= block[63   :   0];
      end else if (next && (w_ctr_reg > 15)) begin
        w_mem[00] <= w_mem[01]; w_mem[01] <= w_mem[02]; w_mem[02] <= w_mem[03];
        w_mem[03] <= w_mem[04]; w_mem[04] <= w_mem[05]; w_mem[05] <= w_mem[06];
        w_mem[06] <= w_mem[07]; w_mem[07] <= w_mem[08]; w_mem[08] <= w_mem[09];
        w_mem[09] <= w_mem[10]; w_mem[10] <= w_mem[11]; w_mem[11] <= w_mem[12];
        w_mem[12] <= w_mem[13]; w_mem[13] <= w_mem[14]; w_mem[14] <= w_mem[15];
        w_mem[15] <= w_new;
      end
      if (w_ctr_we) w_ctr_reg <= w_ctr_new;
    end
  end

  always @* begin : w_ctr_logic
    w_ctr_new = 7'h0;
    w_ctr_we  = init | next;
    if (init) w_ctr_new = 7'h00;
    else if (next) w_ctr_new = w_ctr_reg + 7'h01;
  end
endmodule
