`default_nettype none

module sha512_h_constants(
                          input wire  [1 : 0]  mode,

                          output wire [63 : 0] H0,
                          output wire [63 : 0] H1,
                          output wire [63 : 0] H2,
                          output wire [63 : 0] H3,
                          output wire [63 : 0] H4,
                          output wire [63 : 0] H5,
                          output wire [63 : 0] H6,
                          output wire [63 : 0] H7
                         );

  //----------------------------------------------------------------
  // One-hot mode decoding for parallel selection [rtlscout-cutpoints].
  // Restructuring the selection logic from a priority/case chain to 
  // a parallel one-hot structure helps the synthesis tool (ABC) 
  // optimize the constant-muxing logic more effectively.
  //----------------------------------------------------------------
  wire m0 = (mode == 2'b00); // SHA-512/224
  wire m1 = (mode == 2'b01); // SHA-512/256
  wire m2 = (mode == 2'b10); // SHA-384
  wire m3 = (mode == 2'b11); // SHA-512


  //----------------------------------------------------------------
  // Bitwise selection logic for optimal area and timing.
  // By using bitwise AND with one-hot signals and ORing the results,
  // we create a very flat logic structure. Since the inputs are 
  // constants, the synthesis tool will reduce this to simple 
  // gates (AND/OR/INV) for each bit, sharing logic where bits 
  // are identical across modes.
  //----------------------------------------------------------------
  assign H0 = (64'h8c3d37c819544da2 & {64{m0}}) | 
              (64'h22312194fc2bf72c & {64{m1}}) | 
              (64'hcbbb9d5dc1059ed8 & {64{m2}}) | 
              (64'h6a09e667f3bcc908 & {64{m3}});

  assign H1 = (64'h73e1996689dcd4d6 & {64{m0}}) | 
              (64'h9f555fa3c84c64c2 & {64{m1}}) | 
              (64'h629a292a367cd507 & {64{m2}}) | 
              (64'hbb67ae8584caa73b & {64{m3}});

  assign H2 = (64'h1dfab7ae32ff9c82 & {64{m0}}) | 
              (64'h2393b86b6f53b151 & {64{m1}}) | 
              (64'h9159015a3070dd17 & {64{m2}}) | 
              (64'h3c6ef372fe94f82b & {64{m3}});

  assign H3 = (64'h679dd514582f9fcf & {64{m0}}) | 
              (64'h963877195940eabd & {64{m1}}) | 
              (64'h152fecd8f70e5939 & {64{m2}}) | 
              (64'ha54ff53a5f1d36f1 & {64{m3}});

  assign H4 = (64'h0f6d2b697bd44da8 & {64{m0}}) | 
              (64'h96283ee2a88effe3 & {64{m1}}) | 
              (64'h67332667ffc00b31 & {64{m2}}) | 
              (64'h510e527fade682d1 & {64{m3}});

  assign H5 = (64'h77e36f7304c48942 & {64{m0}}) | 
              (64'hbe5e1e2553863992 & {64{m1}}) | 
              (64'h8eb44a8768581511 & {64{m2}}) | 
              (64'h9b05688c2b3e6c1f & {64{m3}});

  assign H6 = (64'h3f9d85a86a1d36c8 & {64{m0}}) | 
              (64'h2b0199fc2c85b8aa & {64{m1}}) | 
              (64'hdb0c2e0d64f98fa7 & {64{m2}}) | 
              (64'h1f83d9abfb41bd6b & {64{m3}});

  assign H7 = (64'h1112e6ad91d692a1 & {64{m0}}) | 
              (64'h0eb72ddc81c52ca2 & {64{m1}}) | 
              (64'h47b5481dbefa4fa4 & {64{m2}}) | 
              (64'h5be0cd19137e2179 & {64{m3}});

endmodule // sha512_h_constants
