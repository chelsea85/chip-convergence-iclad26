// FILE: prim_subreg_shadow.v
module prim_subreg_shadow (
	clk_i,
	rst_ni,
	rst_shadowed_ni,
	re,
	we,
	wd,
	de,
	d,
	qe,
	q,
	ds,
	qs,
	phase,
	err_update,
	err_storage
);
	reg _sv2v_0;
	parameter signed [31:0] DW = 32;
	parameter [2:0] SwAccess = 3'd0;
	parameter [DW - 1:0] RESVAL = 1'sb0;
	parameter [0:0] Mubi = 1'b0;
	input clk_i;
	input rst_ni;
	input rst_shadowed_ni;
	input re;
	input we;
	input [DW - 1:0] wd;
	input de;
	input [DW - 1:0] d;
	output wire qe;
	output wire [DW - 1:0] q;
	output wire [DW - 1:0] ds;
	output wire [DW - 1:0] qs;
	output wire phase;
	output wire err_update;
	output wire err_storage;

	localparam [2:0] InvertedSwAccess = (SwAccess == 3'd4 ? 3'd5 : (SwAccess == 3'd5 ? 3'd4 : SwAccess));
	
	wire phase_clear;
	reg phase_q;
	reg shadow_we;
	reg [DW - 1:0] shadow_wd;
	reg [DW - 1:0] shadow_q;
	
	wire [DW - 1:0] committed_q;
	wire [DW - 1:0] committed_qs;
	wire committed_qe;
	wire wr_en;
	wire [DW - 1:0] wr_data;

	// Primary arbitration for write enable and data
	prim_subreg_arb #(
		.DW(DW),
		.SwAccess(SwAccess),
		.Mubi(Mubi)
	) wr_en_data_arb(
		.we(we),
		.wd(wd),
		.de(de),
		.d(d),
		.q(committed_q),
		.wr_en(wr_en),
		.wr_data(wr_data)
	);

	// Optimization: Simplify phase_clear logic
	assign phase_clear = (SwAccess == 3'd1 ? 1'b0 : re);

	// Optimization: Use cutpoints for wide comparisons to help synthesis partitioning
	// [rtlscout-cutpoints]
	wire shadow_match_wr_data = (shadow_q == ~wr_data);
	wire shadow_match_committed = (shadow_q == ~committed_q);
	
	// err_update and err_storage are mutually exclusive based on phase_q
	assign err_update = phase_q & wr_en & !shadow_match_wr_data;
	assign err_storage = !phase_q & !shadow_match_committed;

	// Optimization: Simplified phase register logic
	// phase_q toggles on valid write in phase 0, and resets on any write/error in phase 1
	always @(posedge clk_i or negedge rst_ni) begin : phase_reg
		if (!rst_ni)
			phase_q <= 1'b0;
		else
			phase_q <= !phase_q & wr_en & !err_storage & !phase_clear;
	end

	// Shadow register storage
	always @(posedge clk_i or negedge rst_shadowed_ni)
		if (!rst_shadowed_ni)
			shadow_q <= ~RESVAL;
		else if (shadow_we)
			shadow_q <= shadow_wd;

	generate
		if (InvertedSwAccess == SwAccess) begin : gen_shadow_reg_std
			wire shadow_wr_en;
			wire [DW - 1:0] shadow_wr_data;
			prim_subreg_arb #(
				.DW(DW),
				.SwAccess(InvertedSwAccess),
				.Mubi(Mubi)
			) wr_en_data_arb_shadow(
				.we(we),
				.wd(~wr_data),
				.de(de),
				.d(~d),
				.q(shadow_q),
				.wr_en(shadow_wr_en),
				.wr_data(shadow_wr_data)
			);

			// Optimization: Flattened shadow control logic
			wire update_or_clear = err_update | phase_clear;
			always @(*) begin
				shadow_we = !err_storage & (update_or_clear | (!phase_q & shadow_wr_en));
				shadow_wd = update_or_clear ? ~committed_q : shadow_wr_data;
			end
		end
		else begin : gen_shadow_reg_wxx
			wire shadow_wr_en_p0, shadow_wr_en_p1;
			wire [DW - 1:0] shadow_wr_data_p0, shadow_wr_data_p1;
			
			prim_subreg_arb #(
				.DW(DW),
				.SwAccess(3'd0),
				.Mubi(Mubi)
			) wr_en_data_arb_p0(
				.we(we),
				.wd(~wd),
				.de(de),
				.d(~d),
				.q(shadow_q),
				.wr_en(shadow_wr_en_p0),
				.wr_data(shadow_wr_data_p0)
			);
			
			prim_subreg_arb #(
				.DW(DW),
				.SwAccess(InvertedSwAccess),
				.Mubi(Mubi)
			) wr_en_data_arb_p1(
				.we(we),
				.wd(~wr_data),
				.de(de),
				.d(~d),
				.q(~committed_q),
				.wr_en(shadow_wr_en_p1),
				.wr_data(shadow_wr_data_p1)
			);

			wire update_or_clear = err_update | phase_clear;
			always @(*) begin
				shadow_we = !err_storage & (update_or_clear | (phase_q ? shadow_wr_en_p1 : shadow_wr_en_p0));
				shadow_wd = update_or_clear ? ~committed_q : (phase_q ? shadow_wr_data_p1 : shadow_wr_data_p0);
			end
		end
	endgenerate

	// Optimization: Simplified committed register enables
	// Since err_storage is 0 when phase_q is 1, we can simplify the logic
	wire committed_we = we & phase_q & !err_update;
	wire committed_de = de & phase_q & !err_update;

	prim_subreg #(
		.DW(DW),
		.SwAccess(SwAccess),
		.RESVAL(RESVAL),
		.Mubi(Mubi)
	) committed_reg(
		.clk_i(clk_i),
		.rst_ni(rst_ni),
		.we(committed_we),
		.wd(wr_data),
		.de(committed_de),
		.d(d),
		.qe(committed_qe),
		.q(committed_q),
		.ds(ds),
		.qs(committed_qs)
	);

	assign phase = phase_q;
	assign qe = committed_qe;
	assign q = committed_q;
	assign qs = committed_qs;

	initial _sv2v_0 = 0;
endmodule
