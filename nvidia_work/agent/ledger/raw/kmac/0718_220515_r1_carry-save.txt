// FILE: tlul_cmd_intg_chk.v
module tlul_cmd_intg_chk (
	tl_i,
	err_o

	localparam signed [31:0] tlul_pkg_DataIntgWidth = 7;

	localparam signed [31:0] top_pkg_TL_AUW = 23;
	localparam signed [31:0] tlul_pkg_RsvdWidth = ((top_pkg_TL_AUW - prim_mubi_pkg_MuBi4Width) - tlul_pkg_H2DCmdIntgWidth) - tlul_pkg_DataIntgWidth;
	localparam signed [31:0] top_pkg_TL_AIW = 8;
	localparam signed [31:0] top_pkg_TL_AW = 32;
	localparam signed [31:0] top_pkg_TL_DW = 32;
	localparam signed [31:0] top_pkg_TL_DBW = top_pkg_TL_DW >> 3;
	localparam signed [31:0] top_pkg_TL_SZW = $clog2($clog2(top_pkg_TL_DBW) + 1);

	// Total width: (7 + 3 + 8 + 32 + 4 + 32) + (5 + 4 + 7 + 7) + 1 = 110 bits
	input wire [109:0] tl_i;
	output wire err_o;

	// [rtlscout-cutpoints] Declare named intermediate wires at logical boundaries
	// to help synthesis partitioning and optimization.
	wire a_valid;
	wire [2:0]  a_opcode;
	wire [31:0] a_address;
	wire [3:0]  a_mask;
	wire [31:0] a_data;
	wire [3:0]  a_instr_type;
	wire [6:0]  a_cmd_intg;
	wire [6:0]  a_data_intg;

	// Explicit bit-slicing based on TL-UL H2D bus structure
	assign a_valid      = tl_i[109];
	assign a_opcode     = tl_i[108:106];
	assign a_address    = tl_i[91:60];
	assign a_mask       = tl_i[59:56];
	assign a_data       = tl_i[55:24];
	assign a_instr_type = tl_i[18:15];
	assign a_cmd_intg   = tl_i[14:8];
	assign a_data_intg  = tl_i[7:1];

	// Reconstruct the command payload for SECDED check.
	// The original function 'tlul_pkg_extract_h2d_cmd_intg' maps:
	// payload[38:7]  <- a_address
	// payload[6:4]   <- a_opcode
	// payload[3:0]   <- a_mask
	// payload[42:39] <- a_instr_type
	wire [42:0] cmd_payload;
	assign cmd_payload = {a_instr_type, a_address, a_opcode, a_mask};

	// SECDED Decoder for Command Integrity
	// Input is {syndrome (7 bits), data (57 bits)}. 
	// The 43-bit cmd_payload is zero-extended to 57 bits.
	wire [1:0] cmd_err;
	prim_secded_inv_64_57_dec u_chk(
		.data_i({a_cmd_intg, 14'b0, cmd_payload}),
		.data_o(),
		.syndrome_o(),
		.err_o(cmd_err)
	);

	// Data Integrity Decoder
	wire data_err;
	tlul_data_integ_dec u_tlul_data_integ_dec(
		.data_intg_i({a_data_intg, a_data}),
		.data_err_o(data_err)
	);

	// Final error signal: only valid if a_valid is high.
	// Reduction OR on cmd_err (2 bits) and data_err (1 bit).
	assign err_o = a_valid & (cmd_err[1] | cmd_err[0] | data_err);

endmodule
