// FILE: tlul_cmd_intg_chk.v
module tlul_cmd_intg_chk (
	tl_i,
	err_o
);
	localparam signed [31:0] prim_mubi_pkg_MuBi4Width = 4;
	localparam signed [31:0] tlul_pkg_DataIntgWidth = 7;
	localparam signed [31:0] tlul_pkg_H2DCmdIntgWidth = 7;
	localparam signed [31:0] top_pkg_TL_AUW = 23;
	localparam signed [31:0] tlul_pkg_RsvdWidth = ((top_pkg_TL_AUW - prim_mubi_pkg_MuBi4Width) - tlul_pkg_H2DCmdIntgWidth) - tlul_pkg_DataIntgWidth;
	localparam signed [31:0] top_pkg_TL_AIW = 8;
	localparam signed [31:0] top_pkg_TL_AW = 32;
	localparam signed [31:0] top_pkg_TL_DW = 32;
	localparam signed [31:0] top_pkg_TL_DBW = top_pkg_TL_DW >> 3;
	localparam signed [31:0] top_pkg_TL_SZW = $clog2($clog2(top_pkg_TL_DBW) + 1);

	// Total width: 7+3+8+32+4+32+23 + 1 = 110 bits
	input wire [109:0] tl_i;
	output wire err_o;

	// Optimization: Replace complex function-based slicing with explicit wires
	// to reduce boolean network depth and help synthesis partitioning.
	// Base offset for a_user fields = 23
	
	// cmd payload extraction (43 bits total)
	wire [42:0] cmd;
	assign cmd[42:39] = tl_i[18:15];   // instr_type
	assign cmd[38:7]  = tl_i[90:59];   // a_address
	assign cmd[6:4]   = tl_i[108:106]; // a_param, a_size, a_source (partial)
	assign cmd[3:0]   = tl_i[58:55];   // a_mask

	// Command Integrity Check (SECDED)
	// Input is {cmd_intg (7b), cmd_payload (57b)}
	wire [1:0] err;
	prim_secded_inv_64_57_dec u_chk (
		.data_i({tl_i[14:8], 14'b0, cmd}),
		.data_o(),
		.syndrome_o(),
		.err_o(err)
	);

	// Data Integrity Check
	// Input is {data_intg (7b), a_data (32b)}
	wire data_err;
	tlul_data_integ_dec u_tlul_data_integ_dec (
		.data_intg_i({tl_i[7:1], tl_i[54:23]}),
		.data_err_o(data_err)
	);

	// Final error output gated by a_valid (tl_i[109])
	// Structure optimized to keep the valid signal at the end of the logic cone.
	assign err_o = tl_i[109] & (|err | data_err);

	// Maintain unused signal for linting
	wire unused_tl;
	assign unused_tl = |tl_i;

endmodule
