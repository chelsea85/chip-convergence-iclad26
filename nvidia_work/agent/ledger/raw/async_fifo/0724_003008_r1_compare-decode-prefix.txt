### Ledger
- **Transform Applied**: `[R-late-arrival-shannon]` (Shannon cofactoring on the late-arriving `winc_cond` control signal).
- **Exact Cone Touched**: `wptr_full` module, specifically the combinational logic cones for `wfull_val` and `awfull_val`.
- **Changed Signals**:
  - `wbinnext`: original width `ADDRSIZE+1` unsigned -> new width `ADDRSIZE+1` unsigned.
  - `wgraynext`: original width `ADDRSIZE+1` unsigned -> new width `ADDRSIZE+1` unsigned.
  - `wfull_val`: original width 1 unsigned -> new width 1 unsigned.
  - `awfull_val`: original width 1 unsigned -> new width 1 unsigned.

`timescale 1 ns / 1 ps
`default_nettype none

module wptr_full

	#(
		parameter ADDRSIZE = 4
	)(
		input  wire                wclk,
		input  wire                wrst_n,
		input  wire                winc,
		input  wire [ADDRSIZE  :0] wq2_rptr,
		output reg                 wfull,
		output reg                 awfull,
		output wire [ADDRSIZE-1:0] waddr,
		output reg  [ADDRSIZE  :0] wptr
	);

    reg  [ADDRSIZE:0] wbin;
    wire [ADDRSIZE:0] wgraynext, wbinnext;
    wire              awfull_val, wfull_val;

	// GRAYSTYLE2 pointer
	always @(posedge wclk or negedge wrst_n) begin

		if (!wrst_n)
			{wbin, wptr} <= 0;
		else
			{wbin, wptr} <= {wbinnext, wgraynext};

	end

    // Memory write-address pointer (okay to use binary to address memory)
    assign waddr = wbin[ADDRSIZE-1:0];

    // Late-arriving control condition
    wire winc_cond = winc & ~wfull;

    // Pre-computed gray-code pointer values for Shannon cofactoring
    wire [ADDRSIZE:0] wgray_next_0 = wptr;
    wire [ADDRSIZE:0] wgray_next_1 = ((wbin + 1) >> 1) ^ (wbin + 1);
    wire [ADDRSIZE:0] wgray_next_2 = ((wbin + 2) >> 1) ^ (wbin + 2);

    assign wbinnext  = winc_cond ? (wbin + 1) : wbin;
    assign wgraynext = winc_cond ? wgray_next_1 : wgray_next_0;

    // Decoded synchronized read pointer target
    wire [ADDRSIZE:0] wq2_rptr_dec = {~wq2_rptr[ADDRSIZE:ADDRSIZE-1], wq2_rptr[ADDRSIZE-2:0]};

    // Parallel comparison branches
    wire wfull_val_0 = (wgray_next_0 == wq2_rptr_dec);
    wire wfull_val_1 = (wgray_next_1 == wq2_rptr_dec);

    wire awfull_val_0 = (wgray_next_1 == wq2_rptr_dec);
    wire awfull_val_1 = (wgray_next_2 == wq2_rptr_dec);

    // Shannon-cofactored selection using the late-arriving control signal
    assign wfull_val  = winc_cond ? wfull_val_1 : wfull_val_0;
    assign awfull_val = winc_cond ? awfull_val_1 : awfull_val_0;

     always @(posedge wclk or negedge wrst_n) begin

        if (!wrst_n) begin
            awfull <= 1'b0;
            wfull  <= 1'b0;
        end else begin
            awfull <= awfull_val;
            wfull  <= wfull_val;
        end
    end

endmodule

`resetall
