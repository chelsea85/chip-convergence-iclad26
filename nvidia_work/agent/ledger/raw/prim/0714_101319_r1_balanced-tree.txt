// FILE: prim_crc32.v
module prim_crc32 (
	clk_i,
	rst_ni,
	set_crc_i,
	crc_in_i,
	data_valid_i,
	data_i,
	crc_out_o
);
	reg _sv2v_0;
	parameter [31:0] BytesPerWord = 4;
	input wire clk_i;
	input wire rst_ni;
	input wire set_crc_i;
	input wire [31:0] crc_in_i;
	input wire data_valid_i;
	input wire [(BytesPerWord * 8) - 1:0] data_i;
	output wire [31:0] crc_out_o;

	// Optimized CRC32 byte calculation using GF(2) linearity.
	// Instead of a 256-entry case statement (which synthesizes to a large MUX tree),
	// we use the property that CRC(b) = XOR(b[i] ? CRC(1 << i) : 0).
	// This results in a much shallower XOR tree, improving timing and area.
	function automatic [31:0] crc32_byte_calc;
		input reg [7:0] b;
		begin
			crc32_byte_calc = (b[0] ? 32'h77073096 : 32'h0) ^
			                  (b[1] ? 32'hee0e612c : 32'h0) ^
			                  (b[2] ? 32'h076dc419 : 32'h0) ^
			                  (b[3] ? 32'h0edb8832 : 32'h0) ^
			                  (b[4] ? 32'h1db71064 : 32'h0) ^
			                  (b[5] ? 32'h3b6e20c8 : 32'h0) ^
			                  (b[6] ? 32'h76dc4190 : 32'h0) ^
			                  (b[7] ? 32'hedb88320 : 32'h0);
		end
	endfunction

	reg [31:0] crc_d;
	reg [31:0] crc_q;
	wire crc_en;
	wire [31:0] crc_stages [0:BytesPerWord];

	assign crc_en = set_crc_i | data_valid_i;
	assign crc_stages[0] = crc_q;

	genvar _gv_i_1;
	generate
		for (_gv_i_1 = 0; _gv_i_1 < BytesPerWord; _gv_i_1 = _gv_i_1 + 1) begin : g_crc_stages
			localparam i = _gv_i_1;
			// The serial dependency here is now much faster because crc32_byte_calc is a shallow XOR tree.
			assign crc_stages[i + 1] = {8'h00, crc_stages[i][31:8]} ^ crc32_byte_calc(crc_stages[i][7:0] ^ data_i[i * 8+:8]);
		end
	endgenerate

	always @(*) begin
		if (_sv2v_0)
			;
		if (set_crc_i)
			crc_d = ~crc_in_i;
		else
			crc_d = crc_stages[BytesPerWord];
	end

	always @(posedge clk_i or negedge rst_ni)
		if (!rst_ni)
			crc_q <= 32'hffffffff;
		else if (crc_en)
			crc_q <= crc_d;

	assign crc_out_o = ~crc_q;

	initial _sv2v_0 = 0;
endmodule
